# Forge API Implementation Plan (plan 1 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the Forge backend to `ember_api`: worker pairing, the worker WebSocket, group chats, routing, permission relay, and the HTTP/SSE routes the Forge page will use.

**Architecture:** A `ForgeRuntime` owns long-lived components (`WorkerRegistry`, `GroupEventBus`, `MessageStore`, `TurnCoordinator`). Account-scoped services (`WorkerPairingService`, `GroupService`) take a request session and an `account_id`. Routes stay thin. A worker connects out over WebSocket with a bearer pairing token; the coordinator turns user messages into `turn` frames and worker frames into stored messages and browser events.

**Tech Stack:** Python 3.11+, FastAPI (+ WebSocket), SQLAlchemy 2 async on SQLite, Alembic, pydantic v2, pytest. New dependency: `websockets`.

**Spec:** `docs/superpowers/specs/2026-10-18-forge-design.md` (read its "Wire protocol (exact)" section first). Later plans: `2026-10-18-forge-worker.md`, `2026-10-18-forge-web.md`.

```python
"""Shared setup for the Forge tests: a real SQLite database with two accounts (alice = 1, bob = 2)."""

from __future__ import annotations

import asyncio
import json
import uuid
from collections.abc import Awaitable, Callable
from pathlib import Path
from typing import TypeVar

from src.db import Database
from src.models import Account, ForgeGroup, ForgeGroupWorker, ForgeWorker

T = TypeVar("T")


class ForgeEnv:
    def __init__(self, tmp_path: Path) -> None:
        self.database = Database(f"sqlite+aiosqlite:///{(tmp_path / 'forge.db').as_posix()}")

    async def setup(self) -> None:
        await self.database.create_tables()
        async with self.database.sessions() as session:
            for name in ("alice", "bob"):
                session.add(Account(username=name, email=f"{name}@example.com", password_hash="x"))
            await session.commit()

    async def add_worker(
        self, account_id: int = 1, provider: str = "claude", label: str | None = None, plan: str | None = "Max"
    ) -> ForgeWorker:
        async with self.database.sessions() as session:
            worker = ForgeWorker(
                account_id=account_id,
                provider=provider,
                label=label or provider,
                token_hash=uuid.uuid4().hex + uuid.uuid4().hex,
                reported_plan=plan,
            )
            session.add(worker)
            await session.commit()
            return worker

    async def allow(self, provider: str, *plans: str) -> None:
        from src.services.forge.plans import ClaudePlanPolicy, CodexPlanPolicy

        async with self.database.sessions() as session:
            policy = ClaudePlanPolicy(session) if provider == "claude" else CodexPlanPolicy(session)
            await policy.set_allowed(list(plans))

    async def add_group(
        self, account_id: int = 1, workers: tuple[ForgeWorker, ...] | list[ForgeWorker] = (), mode: str = "ask", relay: bool = False
    ) -> ForgeGroup:
        async with self.database.sessions() as session:
            group = ForgeGroup(account_id=account_id, title="Build", mode=mode, relay=relay)
            session.add(group)
            await session.flush()
            for worker in workers:
                session.add(ForgeGroupWorker(group_id=group.id, worker_id=worker.id, folder="C:/work"))
            await session.commit()
            return group


def run(tmp_path: Path, scenario: Callable[[ForgeEnv], Awaitable[T]]) -> T:
    """Runs `scenario(env)` against a fresh database."""
    env = ForgeEnv(tmp_path)

    async def go() -> T:
        await env.setup()
        try:
            return await scenario(env)
        finally:
            await env.database.dispose()

    return asyncio.run(go())


class FakeTransport:
    """Stands in for a worker's WebSocket: records frames sent and the close code."""

    def __init__(self) -> None:
        self.sent: list[dict] = []
        self.closed: int | None = None

    async def send_text(self, data: str) -> None:
        self.sent.append(json.loads(data))

    async def close(self, code: int = 1000) -> None:
        self.closed = code


async def settle(predicate: Callable[[], object], timeout: float = 3.0) -> None:
    """Waits (without blocking the loop) until `predicate()` is truthy."""
    waited = 0.0
    while not predicate():
        if waited >= timeout:
            raise AssertionError("condition not reached in time")
        await asyncio.sleep(0.01)
        waited += 0.01


async def settle_async(predicate: Callable[[], Awaitable[object]], timeout: float = 3.0) -> None:
    """Like settle, for a predicate that is a coroutine function."""
    waited = 0.0
    while not await predicate():
        if waited >= timeout:
            raise AssertionError("condition not reached in time")
        await asyncio.sleep(0.01)
        waited += 0.01
```

```python
"""Forge: a user's workers (the Claude and Codex CLIs on their own computer) and their group chats."""

from __future__ import annotations

from datetime import datetime

from sqlalchemy import JSON, ForeignKey, String, UniqueConstraint, false
from sqlalchemy.orm import Mapped, mapped_column

from src.db import Base, utcnow


class ForgeWorker(Base):
    """One paired worker. `label` is its @mention handle (lower case, unique per
    account). Only the SHA-256 of the pairing token is kept. Deleting the row
    revokes the worker: its group memberships go with it (foreign key cascade)."""

    __tablename__ = "forge_workers"
    __table_args__ = (UniqueConstraint("account_id", "label"), UniqueConstraint("token_hash"))

    id: Mapped[int] = mapped_column(primary_key=True)
    account_id: Mapped[int] = mapped_column(ForeignKey("accounts.id", ondelete="CASCADE"), index=True)
    provider: Mapped[str] = mapped_column(String(16))
    label: Mapped[str] = mapped_column(String(24))
    token_hash: Mapped[str] = mapped_column(String(64))
    # The plan the worker said its CLI is logged in with; Ember does not verify it.
    reported_plan: Mapped[str | None] = mapped_column(String(80), default=None)
    last_seen: Mapped[datetime | None] = mapped_column(default=None)
    created_at: Mapped[datetime] = mapped_column(default=utcnow)


class ForgeGroup(Base):
    """A group chat: one human (the account) and the workers invited into it."""

    __tablename__ = "forge_groups"

    id: Mapped[int] = mapped_column(primary_key=True)
    account_id: Mapped[int] = mapped_column(ForeignKey("accounts.id", ondelete="CASCADE"), index=True)
    title: Mapped[str] = mapped_column(String(80))
    # ask | accept_edits | plan | full_auto
    mode: Mapped[str] = mapped_column(String(16), default="ask")
    # A worker may hand a turn to another worker (off by default).
    relay: Mapped[bool] = mapped_column(default=False, server_default=false())
    # The worker the user addressed last; a message with no @mention goes there.
    last_worker_id: Mapped[int | None] = mapped_column(default=None)
    created_at: Mapped[datetime] = mapped_column(default=utcnow)


class ForgeGroupWorker(Base):
    """A worker invited into a group, with the folder it works in on its own computer."""

    __tablename__ = "forge_group_workers"

    group_id: Mapped[int] = mapped_column(ForeignKey("forge_groups.id", ondelete="CASCADE"), primary_key=True)
    worker_id: Mapped[int] = mapped_column(ForeignKey("forge_workers.id", ondelete="CASCADE"), primary_key=True)
    folder: Mapped[str] = mapped_column(String(500))
    # The newest message this worker has been shown (the next turn sends only what came after).
    last_seq: Mapped[int] = mapped_column(default=0)


class ForgeMessage(Base):
    """One transcript entry. `seq` rises by one per group, so a browser resumes from the last one it saw."""

    __tablename__ = "forge_messages"
    __table_args__ = (UniqueConstraint("group_id", "seq"),)

    id: Mapped[int] = mapped_column(primary_key=True)
    group_id: Mapped[int] = mapped_column(ForeignKey("forge_groups.id", ondelete="CASCADE"), index=True)
    seq: Mapped[int]
    # "user", "system", or a worker id as text.
    sender: Mapped[str] = mapped_column(String(16))
    # text | tool | permission_request | permission_result | system
    kind: Mapped[str] = mapped_column(String(24))
    body: Mapped[dict] = mapped_column(JSON)
    created_at: Mapped[datetime] = mapped_column(default=utcnow)
```

```python
validated here; frames for a worker are plain dicts built by the helpers below."""

from __future__ import annotations

from typing import Annotated, Literal, Union

from pydantic import BaseModel, Field, TypeAdapter, ValidationError

PROVIDERS = ("claude", "codex")
MODES = ("ask", "accept_edits", "plan", "full_auto")

    type: Literal["hello"] = "hello"
    plan: str | None = Field(default=None, max_length=80)
    version: str = Field(default="", max_length=40)


class Heartbeat(BaseModel):
    type: Literal["heartbeat"] = "heartbeat"


class Delta(BaseModel):
    type: Literal["delta"] = "delta"
    turn_id: str = Field(max_length=64)
    text: str = Field(max_length=20_000)


class ToolUsed(BaseModel):
    type: Literal["tool"] = "tool"
    turn_id: str = Field(max_length=64)
    name: str = Field(max_length=80)
    summary: str = Field(default="", max_length=500)


class PermissionRequested(BaseModel):
    type: Literal["permission_request"] = "permission_request"
    turn_id: str = Field(max_length=64)
    request_id: str = Field(max_length=64)
    tool: str = Field(max_length=80)
    summary: str = Field(default="", max_length=500)


class Usage(BaseModel):
    type: Literal["usage"] = "usage"
```

```python
"""Which reported subscription plans may be invited into a group.

The allowlists are not guessed: an administrator types the plan strings the
workers report (the Forge page shows them) into app_settings. Empty means
nothing is eligible. This is a guard in Ember's UI and API only; the provider
enforces the real limit."""

from __future__ import annotations

import json
from abc import ABC, abstractmethod

from sqlalchemy.ext.asyncio import AsyncSession

from src.db import utcnow
from src.models import AppSetting, ForgeWorker

MAX_PLANS = 12
PLAN_MAX = 40
_SETTING_MAX = 200  # app_settings.value is a String(200)


class InvalidPlanList(ValueError):
    """The allowlist is not acceptable; the message is safe to show."""


def normalize_plan(plan: str) -> str:
    return " ".join(plan.split()).lower()


class PlanPolicy(ABC):
    """The plans one provider's workers may be invited with."""

    provider: str

    @abstractmethod
    async def allowed(self) -> list[str]:
        """Normalized plan strings."""

    async def is_allowed(self, reported_plan: str | None) -> bool:
        if not reported_plan:
            return False
        return normalize_plan(reported_plan) in await self.allowed()


class StoredPlanPolicy(PlanPolicy):
    """The allowlist is a JSON list in app_settings under `forge_plans_<provider>`."""

    def __init__(self, session: AsyncSession) -> None:
        self._session = session

    @property
    def _name(self) -> str:
        return f"forge_plans_{self.provider}"

    async def allowed(self) -> list[str]:
        row = await self._session.get(AppSetting, self._name)
        if row is None:
            return []
        try:
            value = json.loads(row.value)
        except ValueError:
            return []
        if not isinstance(value, list):
            return []
        return [normalize_plan(p) for p in value if isinstance(p, str)]

    async def set_allowed(self, plans: list[str]) -> list[str]:
        """Saves the list (cleaned: lower case, no blanks, no repeats) and returns it."""
        cleaned = list(dict.fromkeys(n for n in (normalize_plan(p) for p in plans) if n))
        if len(cleaned) > MAX_PLANS:
            raise InvalidPlanList(f"At most {MAX_PLANS} plans")
        if any(len(p) > PLAN_MAX for p in cleaned):
            raise InvalidPlanList(f"A plan name is at most {PLAN_MAX} characters")
        text = json.dumps(cleaned)
        if len(text) > _SETTING_MAX:
            raise InvalidPlanList("The plan list is too long")
        row = await self._session.get(AppSetting, self._name)
        if row is None:
            self._session.add(AppSetting(name=self._name, value=text))
        else:
            row.value = text
            row.updated_at = utcnow()
        await self._session.commit()
        return cleaned


class ClaudePlanPolicy(StoredPlanPolicy):
    provider = "claude"


class CodexPlanPolicy(StoredPlanPolicy):
    provider = "codex"


class PlanPolicyRegistry:
    """Maps a provider to its policy. A new provider is one new policy subclass listed here."""

    def __init__(self, session: AsyncSession) -> None:
        self._policies: dict[str, StoredPlanPolicy] = {
            policy.provider: policy for policy in (ClaudePlanPolicy(session), CodexPlanPolicy(session))
        }

    def for_provider(self, provider: str) -> StoredPlanPolicy:
        try:
            return self._policies[provider]
        except KeyError:
            raise ValueError(f"Unknown provider: {provider}") from None

    async def allowed(self) -> dict[str, list[str]]:
        return {provider: await policy.allowed() for provider, policy in self._policies.items()}

    async def problem(self, worker: ForgeWorker) -> str | None:
        """Why the worker cannot be used, or None when it can."""
        policy = self._policies.get(worker.provider)
        if policy is None:
            return f"Unknown provider: {worker.provider}"
        if await policy.is_allowed(worker.reported_plan):
            return None
        return f"Plan not eligible: {worker.reported_plan or 'not reported yet'}"
```

```python
"""Pairing a worker to an account, and recognizing a worker that connects.

A pairing token is shown once. Only its SHA-256 is stored: the token is long and
random, so a plain hash is enough and lookups are by hash."""

from __future__ import annotations

import hashlib
import re
import secrets
from dataclasses import dataclass

from sqlalchemy import delete, func, select, update
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from src.db import utcnow
from src.models import ForgeWorker
from src.services.forge.protocol import PROVIDERS

TOKEN_PREFIX = "ft_"
MAX_WORKERS_PER_ACCOUNT = 10
_LABEL = re.compile(r"^[a-z0-9_-]{1,24}$")
_PLAN_MAX = 80  # forge_workers.reported_plan


class WorkerNotFound(Exception):
    """Unknown id, or another account's (deliberately indistinguishable)."""


class WorkerLimitError(Exception):
    """Too many workers for the account (maps to 409)."""


class WorkerLabelTaken(Exception):
    """Another worker of this account already has the label."""


class InvalidWorkerLabel(ValueError):
    """The label is not a valid @mention handle; the message is safe to show."""


def hash_token(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def clean_label(label: str) -> str:
    cleaned = label.strip().lower()
    if not _LABEL.fullmatch(cleaned):
        raise InvalidWorkerLabel("A label is 1 to 24 letters, digits, - or _")
    return cleaned


@dataclass(frozen=True)
class NewWorker:
    worker: ForgeWorker
    token: str  # shown to the user once


class WorkerPairingService:
    """One account's workers."""

    def __init__(self, session: AsyncSession, account_id: int) -> None:
        self._session = session
        self._account_id = account_id

    async def list(self) -> list[ForgeWorker]:
        rows = await self._session.scalars(
            select(ForgeWorker).where(ForgeWorker.account_id == self._account_id).order_by(ForgeWorker.id)
        )
        return list(rows)

    async def get(self, worker_id: int) -> ForgeWorker:
        worker = await self._session.scalar(
            select(ForgeWorker).where(ForgeWorker.id == worker_id, ForgeWorker.account_id == self._account_id)
        )
        if worker is None:
            raise WorkerNotFound(worker_id)
        return worker

    async def create(self, provider: str, label: str) -> NewWorker:
        if provider not in PROVIDERS:
            raise ValueError(f"Unknown provider: {provider}")
        label = clean_label(label)
        count = await self._session.scalar(
            select(func.count()).select_from(ForgeWorker).where(ForgeWorker.account_id == self._account_id)
        )
        if (count or 0) >= MAX_WORKERS_PER_ACCOUNT:
            raise WorkerLimitError(f"At most {MAX_WORKERS_PER_ACCOUNT} workers per account - remove some first")
        taken = await self._session.scalar(
            select(ForgeWorker.id).where(ForgeWorker.account_id == self._account_id, ForgeWorker.label == label)
        )
        if taken is not None:
            raise WorkerLabelTaken(label)
        token = TOKEN_PREFIX + secrets.token_urlsafe(32)
        worker = ForgeWorker(account_id=self._account_id, provider=provider, label=label, token_hash=hash_token(token))
        self._session.add(worker)
        try:
            await self._session.commit()
        except IntegrityError as error:
            await self._session.rollback()
            raise WorkerLabelTaken(label) from error
        return NewWorker(worker, token)

    async def delete(self, worker_id: int) -> ForgeWorker:
        """Revokes the worker: the row goes, and so do its group memberships."""
        worker = await self.get(worker_id)
        await self._session.execute(delete(ForgeWorker).where(ForgeWorker.id == worker.id))
        await self._session.commit()
        return worker


class WorkerAuthenticator:
    """Recognizes a connecting worker by its token. Not scoped to an account."""

    def __init__(self, session: AsyncSession) -> None:
        self._session = session

    async def authenticate(self, token: str) -> ForgeWorker | None:
        if not token.startswith(TOKEN_PREFIX):
            return None
        return await self._session.scalar(select(ForgeWorker).where(ForgeWorker.token_hash == hash_token(token)))

    async def record_hello(self, worker_id: int, plan: str | None) -> None:
        cleaned = " ".join(plan.split())[:_PLAN_MAX] if plan else None
        await self._session.execute(
            update(ForgeWorker).where(ForgeWorker.id == worker_id).values(reported_plan=cleaned or None, last_seen=utcnow())
        )
        await self._session.commit()

    async def touch(self, worker_id: int) -> None:
        await self._session.execute(update(ForgeWorker).where(ForgeWorker.id == worker_id).values(last_seen=utcnow()))
        await self._session.commit()
```

```python
"""Live worker sockets. A connection wraps anything that can send text and close, so the
services never touch FastAPI's WebSocket type and tests use a fake."""

from __future__ import annotations

import asyncio
import json
from typing import Protocol


class Transport(Protocol):
    async def send_text(self, data: str) -> None: ...

    async def close(self, code: int = 1000) -> None: ...


class WorkerConnection:
    """One live worker socket. Sends are serialized so frames never interleave."""

    def __init__(self, worker_id: int, transport: Transport) -> None:
        self.worker_id = worker_id
        self._transport = transport
        self._lock = asyncio.Lock()

    async def send(self, frame: dict) -> None:
        async with self._lock:
            await self._transport.send_text(json.dumps(frame, ensure_ascii=False))

    async def close(self, code: int = 1000) -> None:
        try:
            await self._transport.close(code)
        except Exception:  # already closed or broken: nothing left to do
            pass


class WorkerRegistry:
    """Which workers are online right now: worker id to its newest connection."""

    def __init__(self) -> None:
        self._connections: dict[int, WorkerConnection] = {}

    def register(self, connection: WorkerConnection) -> WorkerConnection | None:
        """Returns the connection this one replaced (the caller closes it), if any."""
        replaced = self._connections.get(connection.worker_id)
        self._connections[connection.worker_id] = connection
        return replaced

    def get(self, worker_id: int) -> WorkerConnection | None:
        return self._connections.get(worker_id)

    def drop(self, worker_id: int, connection: WorkerConnection) -> None:
        """Forgets `connection`, unless a newer one has taken its place."""
        if self._connections.get(worker_id) is connection:
            del self._connections[worker_id]

    def is_online(self, worker_id: int) -> bool:
        return worker_id in self._connections
```

```python
"""One account's Forge groups: titles, modes, and which of its workers are invited.

Every lookup filters by account, so another account's id behaves like a missing one."""

from __future__ import annotations

from dataclasses import dataclass

from sqlalchemy import delete, func, select, update
from sqlalchemy.ext.asyncio import AsyncSession

from src.models import ForgeGroup, ForgeGroupWorker, ForgeWorker
from src.services.forge.plans import PlanPolicyRegistry
from src.services.forge.protocol import MODES
from src.services.forge.workers import WorkerNotFound

TITLE_MAX = 80
FOLDER_MAX = 500
MAX_GROUPS_PER_ACCOUNT = 50
MAX_MEMBERS = 8


class GroupNotFound(Exception):
    """Unknown id, or another account's (deliberately indistinguishable)."""


class GroupLimitError(Exception):
    """Too many groups for the account (maps to 409)."""


class MemberLimitError(Exception):
    """The group has as many workers as it can hold (maps to 409)."""


class WorkerNotEligible(Exception):
    """The worker cannot be invited; `reason` is safe to show."""

    def __init__(self, reason: str) -> None:
        super().__init__(reason)
        self.reason = reason


class AlreadyMember(Exception):
    """The worker is already in the group."""


class NotAMember(Exception):
    """The worker is not in the group."""


class InvalidGroup(ValueError):
    """A title, mode or folder is not acceptable; the message is safe to show."""


@dataclass(frozen=True)
class MemberRow:
    worker: ForgeWorker
    folder: str
    last_seq: int


def _clean_title(title: str) -> str:
    cleaned = " ".join(title.split())
    if not cleaned:
        raise InvalidGroup("The title must not be blank")
    if len(cleaned) > TITLE_MAX:
        raise InvalidGroup(f"The title is at most {TITLE_MAX} characters")
    return cleaned


def _clean_mode(mode: str) -> str:
    if mode not in MODES:
        raise InvalidGroup(f"The mode must be one of: {', '.join(MODES)}")
    return mode


def _clean_folder(folder: str) -> str:
    cleaned = folder.strip()
    if not cleaned or "\x00" in cleaned:
        raise InvalidGroup("The folder must not be blank")
    if len(cleaned) > FOLDER_MAX:
        raise InvalidGroup(f"The folder is at most {FOLDER_MAX} characters")
    return cleaned


class GroupService:
    def __init__(self, session: AsyncSession, account_id: int, plans: PlanPolicyRegistry) -> None:
        self._session = session
        self._account_id = account_id
        self._plans = plans

    async def list(self) -> list[ForgeGroup]:
        """Newest first."""
        rows = await self._session.scalars(
            select(ForgeGroup).where(ForgeGroup.account_id == self._account_id).order_by(ForgeGroup.id.desc())
        )
        return list(rows)

    async def get(self, group_id: int) -> ForgeGroup:
        group = await self._session.scalar(
            select(ForgeGroup).where(ForgeGroup.id == group_id, ForgeGroup.account_id == self._account_id)
        )
        if group is None:
            raise GroupNotFound(group_id)
        return group

    async def create(self, title: str, mode: str = "ask") -> ForgeGroup:
        title, mode = _clean_title(title), _clean_mode(mode)
        count = await self._session.scalar(
            select(func.count()).select_from(ForgeGroup).where(ForgeGroup.account_id == self._account_id)
        )
        if (count or 0) >= MAX_GROUPS_PER_ACCOUNT:
            raise GroupLimitError(f"At most {MAX_GROUPS_PER_ACCOUNT} groups per account - delete some first")
        group = ForgeGroup(account_id=self._account_id, title=title, mode=mode)
        self._session.add(group)
        await self._session.commit()
        return group

    async def update(
        self, group_id: int, *, title: str | None = None, mode: str | None = None, relay: bool | None = None
    ) -> ForgeGroup:
        group = await self.get(group_id)
        if title is not None:
            group.title = _clean_title(title)
        if mode is not None:
            group.mode = _clean_mode(mode)
        if relay is not None:
            group.relay = relay
        await self._session.commit()
        return group

    async def delete(self, group_id: int) -> None:
        """Deletes the group; its members and messages go with it (foreign key cascade)."""
        group = await self.get(group_id)
        await self._session.execute(delete(ForgeGroup).where(ForgeGroup.id == group.id))
        await self._session.commit()

    async def members(self, group_id: int) -> list[MemberRow]:
        await self.get(group_id)
        rows = await self._session.execute(
            select(ForgeWorker, ForgeGroupWorker.folder, ForgeGroupWorker.last_seq)
            .join(ForgeGroupWorker, ForgeGroupWorker.worker_id == ForgeWorker.id)
            .where(ForgeGroupWorker.group_id == group_id)
            .order_by(ForgeWorker.id)
        )
        return [MemberRow(worker, folder, last_seq) for worker, folder, last_seq in rows]

    async def member(self, group_id: int, worker_id: int) -> MemberRow:
        for row in await self.members(group_id):
            if row.worker.id == worker_id:
                return row
        raise NotAMember(worker_id)

    async def invite(self, group_id: int, worker_id: int, folder: str) -> MemberRow:
        await self.get(group_id)
        worker = await self._session.scalar(
            select(ForgeWorker).where(ForgeWorker.id == worker_id, ForgeWorker.account_id == self._account_id)
        )
        if worker is None:
            raise WorkerNotFound(worker_id)
        folder = _clean_folder(folder)
        problem = await self._plans.problem(worker)
        if problem is not None:
            raise WorkerNotEligible(problem)
        current = await self.members(group_id)
        if any(row.worker.id == worker_id for row in current):
            raise AlreadyMember(worker_id)
        if len(current) >= MAX_MEMBERS:
            raise MemberLimitError(f"At most {MAX_MEMBERS} workers per group")
        self._session.add(ForgeGroupWorker(group_id=group_id, worker_id=worker_id, folder=folder, last_seq=0))
        await self._session.commit()
        return MemberRow(worker, folder, 0)

    async def remove_member(self, group_id: int, worker_id: int) -> None:
        await self.member(group_id, worker_id)
        await self._session.execute(
            delete(ForgeGroupWorker).where(ForgeGroupWorker.group_id == group_id, ForgeGroupWorker.worker_id == worker_id)
        )
        await self._session.commit()

    async def mark_seen(self, group_id: int, worker_id: int, seq: int) -> None:
        """The worker has now seen messages up to `seq`, and was the last one addressed."""
        await self._session.execute(
            update(ForgeGroupWorker)
            .where(ForgeGroupWorker.group_id == group_id, ForgeGroupWorker.worker_id == worker_id)
            .values(last_seq=seq)
        )
        await self._session.execute(
            update(ForgeGroup)
            .where(ForgeGroup.id == group_id, ForgeGroup.account_id == self._account_id)
            .values(last_worker_id=worker_id)
        )
        await self._session.commit()

    async def problem(self, worker: ForgeWorker) -> str | None:
        """Why the worker cannot take a turn right now (its plan), or None."""
        return await self._plans.problem(worker)
```

```python
"""The transcript. Long-lived (it owns the per-group locks that keep `seq` gapless), so it
opens its own short sessions. It does not check ownership: callers verify the group with
GroupService first."""

from __future__ import annotations

import asyncio

from sqlalchemy import func, select

from src.db import Database
from src.models import ForgeMessage

LIST_LIMIT = 500
HISTORY_LIMIT = 40


def message_event(message: ForgeMessage) -> dict:
    """A stored message as the browser event (spec: Wire protocol, browser events)."""
    return {
        "type": "message",
        "seq": message.seq,
        "sender": message.sender,
        "kind": message.kind,
        "body": message.body,
        "created_at": message.created_at.isoformat(),
    }


class MessageStore:
    def __init__(self, database: Database) -> None:
        self._database = database
        self._locks: dict[int, asyncio.Lock] = {}

    def _lock(self, group_id: int) -> asyncio.Lock:
        return self._locks.setdefault(group_id, asyncio.Lock())

    async def append(self, group_id: int, sender: str, kind: str, body: dict) -> ForgeMessage:
        async with self._lock(group_id):
            async with self._database.sessions() as session:
                last = await session.scalar(
                    select(func.coalesce(func.max(ForgeMessage.seq), 0)).where(ForgeMessage.group_id == group_id)
                )
                message = ForgeMessage(group_id=group_id, seq=(last or 0) + 1, sender=sender, kind=kind, body=body)
                session.add(message)
                await session.commit()
                return message

    async def list_after(self, group_id: int, after: int = 0, limit: int = LIST_LIMIT) -> list[ForgeMessage]:
        async with self._database.sessions() as session:
            rows = await session.scalars(
                select(ForgeMessage)
                .where(ForgeMessage.group_id == group_id, ForgeMessage.seq > after)
                .order_by(ForgeMessage.seq)
                .limit(limit)
            )
            return list(rows)

    async def history(
        self, group_id: int, after_seq: int, *, exclude_seq: int | None = None, limit: int = HISTORY_LIMIT
    ) -> list[ForgeMessage]:
        """The newest `limit` text messages after `after_seq`, oldest first."""
        query = select(ForgeMessage).where(
            ForgeMessage.group_id == group_id, ForgeMessage.seq > after_seq, ForgeMessage.kind == "text"
        )
        if exclude_seq is not None:
            query = query.where(ForgeMessage.seq != exclude_seq)
        async with self._database.sessions() as session:
            rows = list(await session.scalars(query.order_by(ForgeMessage.seq.desc()).limit(limit)))
        rows.reverse()
        return rows

    async def last_seq(self, group_id: int) -> int:
        async with self._database.sessions() as session:
            last = await session.scalar(
                select(func.coalesce(func.max(ForgeMessage.seq), 0)).where(ForgeMessage.group_id == group_id)
            )
            return last or 0
```

```python
"""Live events for the browser: an in-memory fan-out per group, and the streamer that joins
the stored transcript to it. Single process, like the rest of Forge."""

from __future__ import annotations

import asyncio
import json
from collections.abc import AsyncIterator, Callable

from src.services.forge.messages import MessageStore, message_event

QUEUE_MAX = 500


class Subscription:
    """One browser's queue of live events. If it falls too far behind it is told so
    (`lagged`) and ends; the browser reconnects and replays from its last `seq`."""

    def __init__(self, bus: GroupEventBus, group_id: int) -> None:
        self._bus = bus
        self.group_id = group_id
        self._queue: asyncio.Queue[dict] = asyncio.Queue(maxsize=QUEUE_MAX)
        self._lagged = False

    def push(self, event: dict) -> None:
        if self._lagged:
            return
        try:
            self._queue.put_nowait(event)
        except asyncio.QueueFull:
            self._lagged = True

    async def events(self, ping_seconds: float = 20.0) -> AsyncIterator[dict]:
        while True:
            if self._lagged and self._queue.empty():
                yield {"type": "lagged"}
                return
            try:
                event = await asyncio.wait_for(self._queue.get(), ping_seconds)
            except asyncio.TimeoutError:
                yield {"type": "ping"}
                continue
            yield event

    def close(self) -> None:
        self._bus._remove(self)


class GroupEventBus:
    def __init__(self) -> None:
        self._subscriptions: dict[int, set[Subscription]] = {}

    def subscribe(self, group_id: int) -> Subscription:
        subscription = Subscription(self, group_id)
        self._subscriptions.setdefault(group_id, set()).add(subscription)
        return subscription

    def publish(self, group_id: int, event: dict) -> None:
        for subscription in tuple(self._subscriptions.get(group_id, ())):
            subscription.push(event)

    def _remove(self, subscription: Subscription) -> None:
        group = self._subscriptions.get(subscription.group_id)
        if group is not None:
            group.discard(subscription)
            if not group:
                del self._subscriptions[subscription.group_id]


def sse_frame(event: dict) -> bytes:
    """One server-sent-events frame. Stored messages carry `id: <seq>` so a browser can resume."""
    if event["type"] == "ping":
        return b": ping\n\n"
    prefix = f"id: {event['seq']}\n" if event["type"] == "message" else ""
    return f"{prefix}data: {json.dumps(event, ensure_ascii=False)}\n\n".encode()


class GroupStreamer:
    """What one browser sees: stored messages after `after`, the turn state, then live events."""

    def __init__(self, bus: GroupEventBus, store: MessageStore, is_running: Callable[[int], bool]) -> None:
        self._bus = bus
        self._store = store
        self._is_running = is_running

    async def stream(self, group_id: int, after: int, ping_seconds: float = 20.0) -> AsyncIterator[dict]:
        # Subscribe first, so a message stored while we replay is not missed (it is deduplicated by seq).
        subscription = self._bus.subscribe(group_id)
        try:
            last = after
            for message in await self._store.list_after(group_id, after):
                last = message.seq
                yield message_event(message)
            yield {"type": "turn", "running": self._is_running(group_id)}
            async for event in subscription.events(ping_seconds):
                if event["type"] == "message":
                    if event["seq"] <= last:
                        continue
                    last = event["seq"]
                yield event
                if event["type"] == "lagged":
                    return
        finally:
            subscription.close()
```

```python
"""Who gets a message: @mentions from the user, and (when relay is on) a worker handing a
turn to another worker with an @mention on the last line of its reply."""

from __future__ import annotations

import re
from dataclasses import dataclass

# An @handle that is not part of a longer word, address or handle.
_MENTION = re.compile(r"(?<![\w@])@([A-Za-z0-9_-]{1,24})(?![\w@-])")
_HANDOFF = re.compile(r"^@([A-Za-z0-9_-]{1,24})(?![\w@-])[:,]?\s*(.*)$")
_CONTINUE = "Please continue."


@dataclass(frozen=True)
class Member:
    worker_id: int
    label: str


class TurnRouter:
    def __init__(self, max_relay_depth: int = 3) -> None:
        self.max_relay_depth = max_relay_depth

    def target_for_user(self, text: str, members: list[Member], last_worker_id: int | None) -> Member | None:
        by_label = {member.label.lower(): member for member in members}
        for match in _MENTION.finditer(text):
            member = by_label.get(match.group(1).lower())
            if member is not None:
                return member
        for member in members:
            if member.worker_id == last_worker_id:
                return member
        return members[0] if len(members) == 1 else None

    def handoff(self, reply: str, members: list[Member], current_worker_id: int) -> tuple[Member, str] | None:
        """The worker a reply passes the turn to, and what to tell it, or None."""
        lines = [line.strip() for line in reply.strip().splitlines() if line.strip()]
        if not lines:
            return None
        match = _HANDOFF.match(lines[-1])
        if match is None:
            return None
        target = next((m for m in members if m.label.lower() == match.group(1).lower()), None)
        if target is None or target.worker_id == current_worker_id:
            return None
        return target, match.group(2).strip() or _CONTINUE
```

```python
"""Tool-permission requests waiting for the user's Allow or Deny. An unanswered request is
reported to `on_timeout` (the coordinator denies it) after `timeout_seconds`."""

from __future__ import annotations

import asyncio
from collections.abc import Awaitable, Callable
from dataclasses import dataclass


@dataclass(frozen=True)
class PendingPermission:
    request_id: str
    group_id: int
    worker_id: int
    turn_id: str


class PermissionBroker:
    def __init__(
        self, on_timeout: Callable[[PendingPermission], Awaitable[None]], timeout_seconds: float = 600.0
    ) -> None:
        self._on_timeout = on_timeout
        self._timeout = timeout_seconds
        self._pending: dict[str, PendingPermission] = {}
        self._timers: dict[str, asyncio.Task] = {}

    def open(self, pending: PendingPermission) -> None:
        self._pending[pending.request_id] = pending
        self._timers[pending.request_id] = asyncio.create_task(self._expire(pending))

    def resolve(self, group_id: int, request_id: str) -> PendingPermission | None:
        """Takes the request out of the queue; None when it is unknown, answered, or another group's."""
        pending = self._pending.get(request_id)
        if pending is None or pending.group_id != group_id:
            return None
        self._forget(request_id)
        return pending

    def discard_turn(self, turn_id: str) -> list[PendingPermission]:
        dropped = [p for p in self._pending.values() if p.turn_id == turn_id]
        for pending in dropped:
            self._forget(pending.request_id)
        return dropped

    def close(self) -> None:
        for request_id in list(self._pending):
            self._forget(request_id)

    def _forget(self, request_id: str) -> None:
        self._pending.pop(request_id, None)
        timer = self._timers.pop(request_id, None)
        if timer is not None and timer is not asyncio.current_task():
            timer.cancel()

    async def _expire(self, pending: PendingPermission) -> None:
        await asyncio.sleep(self._timeout)
        if self._pending.get(pending.request_id) is None:
            return
        self._forget(pending.request_id)
        await self._on_timeout(pending)
```

```python
"""Runs Forge turns. A user message is routed to a worker as a `turn` frame; the worker's
frames become stored messages (and live browser events); with relay on, a reply that ends
with an @mention hands the turn to another worker. One turn runs per group at a time.

Long-lived and single-process: it opens its own short database sessions."""

from __future__ import annotations

import asyncio
import logging
import uuid
from dataclasses import dataclass, field

from sqlalchemy.ext.asyncio import AsyncSession

from src.db import Database
from src.models import ForgeMessage
from src.services.forge.approvals import PendingPermission, PermissionBroker
from src.services.forge.connections import WorkerRegistry
from src.services.forge.events import GroupEventBus
from src.services.forge.groups import GroupService, MemberRow, NotAMember
from src.services.forge.messages import MessageStore, message_event
from src.services.forge.plans import PlanPolicyRegistry
from src.services.forge.protocol import (
    Delta,
    PermissionRequested,
    ToolUsed,
    TurnDone,
    TurnError,
    Usage,
    WorkerFrame,
    cancel_frame,
    permission_reply_frame,
    turn_frame,
)
from src.services.forge.routing import Member, TurnRouter

logger = logging.getLogger(__name__)

CANCEL_GRACE_SECONDS = 10.0
HISTORY_CHARS = 20_000


class GroupBusy(Exception):
    """A turn is already running in the group (maps to 409)."""


@dataclass(frozen=True)
class TurnOutcome:
    text: str | None = None
    error: str | None = None
    cancelled: bool = False
    usage: dict | None = None


@dataclass
class _ActiveTurn:
    turn_id: str
    group_id: int
    worker_id: int
    label: str
    done: asyncio.Future = field(default_factory=lambda: asyncio.get_running_loop().create_future())
    usage: dict | None = None
    cancelled: bool = False


class TurnCoordinator:
    def __init__(
        self,
        database: Database,
        registry: WorkerRegistry,
        bus: GroupEventBus,
        store: MessageStore,
        router: TurnRouter,
        *,
        permission_timeout: float = 600.0,
        cancel_grace: float = CANCEL_GRACE_SECONDS,
    ) -> None:
        self._database = database
        self._registry = registry
        self._bus = bus
        self._store = store
        self._router = router
        self._cancel_grace = cancel_grace
        self._broker = PermissionBroker(self._deny_expired, permission_timeout)
        # group id -> the task running its chain (None while the claim is being made)
        self._running: dict[int, asyncio.Task | None] = {}
        self._stopping: set[int] = set()
        self._turns: dict[str, _ActiveTurn] = {}
        self._by_worker: dict[int, str] = {}

    # --- what the routes call ------------------------------------------------------

    def is_running(self, group_id: int) -> bool:
        return group_id in self._running

    async def post_user_message(self, account_id: int, group_id: int, text: str) -> ForgeMessage:
        if group_id in self._running:
            raise GroupBusy(group_id)
        self._running[group_id] = None  # claimed before the first await, so two messages cannot both start
        started = False
        try:
            async with self._database.sessions() as session:
                groups = self._groups(session, account_id)
                group = await groups.get(group_id)
                members = [Member(row.worker.id, row.worker.label) for row in await groups.members(group_id)]
            target = self._router.target_for_user(text, members, group.last_worker_id)
            message = await self._post(group_id, "user", "text", {"text": text})
            if target is None:
                await self.post_system(group_id, "No worker was addressed. Start your message with @label.")
                return message
            task = asyncio.create_task(self._run_chain(account_id, group_id, target, text, message.seq))
            self._running[group_id] = task
            task.add_done_callback(lambda finished: self._finished(group_id, finished))
            started = True
            self._publish_turn(group_id, True)
            return message
        finally:
            if not started:
                self._running.pop(group_id, None)

    async def post_system(self, group_id: int, text: str) -> None:
        await self._post(group_id, "system", "system", {"text": text})

    async def resolve_permission(self, account_id: int, group_id: int, request_id: str, allow: bool) -> bool:
        await self._owned_group(account_id, group_id)
        pending = self._broker.resolve(group_id, request_id)
        if pending is None:
            return False
        await self._answer_permission(pending, allow, by="user")
        return True

    async def cancel(self, account_id: int, group_id: int) -> bool:
        """Asks the running worker to stop. False when nothing is running."""
        await self._owned_group(account_id, group_id)
        if group_id not in self._running:
            return False
        self._stopping.add(group_id)  # no further hand-off after this turn
        turn = next((t for t in self._turns.values() if t.group_id == group_id), None)
        if turn is None:
            return True
        turn.cancelled = True
        connection = self._registry.get(turn.worker_id)
        if connection is not None:
            try:
                await connection.send(cancel_frame(turn.turn_id))
            except Exception:
                logger.warning("could not send cancel to worker %s", turn.worker_id)
        # If the worker does not acknowledge in time, the turn is ended here.
        asyncio.get_running_loop().call_later(self._cancel_grace, self._complete, turn, TurnOutcome(cancelled=True))
        return True

    async def shutdown(self) -> None:
        self._broker.close()
        tasks = [task for task in self._running.values() if task is not None]
        for task in tasks:
            task.cancel()
        await asyncio.gather(*tasks, return_exceptions=True)

    # --- what the worker socket calls ----------------------------------------------

    async def on_worker_frame(self, worker_id: int, frame: WorkerFrame) -> None:
        turn_id = getattr(frame, "turn_id", None)
        turn = self._turns.get(turn_id) if turn_id else None
        if turn is None or turn.worker_id != worker_id:
            return  # late, or not this worker's turn
        sender = str(worker_id)
        if isinstance(frame, Delta):
            self._bus.publish(
                turn.group_id,
                {"type": "delta", "turn_id": turn.turn_id, "sender": sender, "label": turn.label, "text": frame.text},
            )
        elif isinstance(frame, ToolUsed):
            body = {"turn_id": turn.turn_id, "label": turn.label, "name": frame.name, "summary": frame.summary}
            await self._post(turn.group_id, sender, "tool", body)
        elif isinstance(frame, PermissionRequested):
            self._broker.open(PendingPermission(frame.request_id, turn.group_id, worker_id, turn.turn_id))
            body = {
                "turn_id": turn.turn_id,
                "label": turn.label,
                "request_id": frame.request_id,
                "tool": frame.tool,
                "summary": frame.summary,
            }
            await self._post(turn.group_id, sender, "permission_request", body)
        elif isinstance(frame, Usage):
            turn.usage = {"input_tokens": frame.input_tokens, "output_tokens": frame.output_tokens}
        elif isinstance(frame, TurnDone):
            self._complete(turn, TurnOutcome(text=frame.text, usage=turn.usage))
        elif isinstance(frame, TurnError):
            self._complete(turn, TurnOutcome(cancelled=True) if turn.cancelled else TurnOutcome(error=frame.message))

    async def on_worker_disconnected(self, worker_id: int) -> None:
        turn = self._turns.get(self._by_worker.get(worker_id, ""))
        if turn is not None:
            self._complete(turn, TurnOutcome(error="worker disconnected"))

    # --- the chain of turns ---------------------------------------------------------

    def _finished(self, group_id: int, task: asyncio.Task) -> None:
        self._running.pop(group_id, None)
        self._stopping.discard(group_id)
        if not task.cancelled() and task.exception() is not None:
            logger.error("forge chain failed", exc_info=task.exception())
        self._publish_turn(group_id, False)

    async def _run_chain(
        self, account_id: int, group_id: int, target: Member, prompt: str, prompt_seq: int | None
    ) -> None:
        depth = 0
        try:
            while True:
                outcome = await self._run_turn(account_id, group_id, target, prompt, prompt_seq)
                if outcome is None or outcome.error or outcome.cancelled or not outcome.text:
                    return
                if group_id in self._stopping:
                    return
                async with self._database.sessions() as session:
                    groups = self._groups(session, account_id)
                    group = await groups.get(group_id)
                    rows = await groups.members(group_id)
                if not group.relay or depth >= self._router.max_relay_depth:
                    return
                handoff = self._router.handoff(
                    outcome.text, [Member(row.worker.id, row.worker.label) for row in rows], target.worker_id
                )
                if handoff is None:
                    return
                (target, prompt), prompt_seq, depth = handoff, None, depth + 1
        except asyncio.CancelledError:
            raise
        except Exception as error:
            logger.exception("forge turn failed")
            try:
                await self.post_system(group_id, f"Forge error: {error}")
            except Exception:
                logger.exception("could not record the failure")

    async def _run_turn(
        self, account_id: int, group_id: int, target: Member, prompt: str, prompt_seq: int | None
    ) -> TurnOutcome | None:
        async with self._database.sessions() as session:
            groups = self._groups(session, account_id)
            group = await groups.get(group_id)
            try:
                member = await groups.member(group_id, target.worker_id)
            except NotAMember:
                await self.post_system(group_id, f"@{target.label} is no longer in this group.")
                return None
            problem = await groups.problem(member.worker)
        label = member.worker.label
        if problem is not None:
            await self.post_system(group_id, f"@{label}: {problem}")
            return None
        connection = self._registry.get(target.worker_id)
        if connection is None:
            await self.post_system(group_id, f"@{label} is offline. Start ember-worker on its computer.")
            return None

        history = await self._history(group_id, member, prompt_seq)
        turn = _ActiveTurn(uuid.uuid4().hex, group_id, target.worker_id, label)
        self._turns[turn.turn_id] = turn
        self._by_worker[turn.worker_id] = turn.turn_id
        try:
            await connection.send(
                turn_frame(
                    turn_id=turn.turn_id,
                    group_id=group_id,
                    folder=member.folder,
                    mode=group.mode,
                    prompt=prompt,
                    history=history,
                )
            )
            outcome = await turn.done
        except Exception as error:  # the socket failed while sending
            outcome = TurnOutcome(error=f"could not reach the worker ({error})")
        finally:
            self._turns.pop(turn.turn_id, None)
            if self._by_worker.get(turn.worker_id) == turn.turn_id:
                del self._by_worker[turn.worker_id]
            self._broker.discard_turn(turn.turn_id)
        await self._finish(account_id, group_id, member, turn, outcome)
        return outcome

    async def _finish(
        self, account_id: int, group_id: int, member: MemberRow, turn: _ActiveTurn, outcome: TurnOutcome
    ) -> None:
        worker, label = member.worker, member.worker.label
        if outcome.error is not None:
            await self.post_system(group_id, f"@{label} failed: {outcome.error}")
            return
        if outcome.cancelled:
            await self.post_system(group_id, f"@{label} stopped.")
            return
        body = {"text": outcome.text, "label": label, "provider": worker.provider, "turn_id": turn.turn_id}
        if outcome.usage:
            body["usage"] = outcome.usage
        await self._post(group_id, str(worker.id), "text", body)
        newest = await self._store.last_seq(group_id)
        async with self._database.sessions() as session:
            await self._groups(session, account_id).mark_seen(group_id, worker.id, newest)

    async def _history(self, group_id: int, member: MemberRow, prompt_seq: int | None) -> list[dict]:
        messages = await self._store.history(group_id, member.last_seq, exclude_seq=prompt_seq)
        lines: list[dict] = []
        total = 0
        for message in reversed(messages):  # newest first, so the budget keeps the latest
            text = message.body.get("text", "")
            total += len(text)
            if total > HISTORY_CHARS:
                break
            sender = "user" if message.sender == "user" else message.body.get("label", message.sender)
            lines.append({"sender": sender, "text": text})
        lines.reverse()
        return lines

    # --- helpers --------------------------------------------------------------------

    def _complete(self, turn: _ActiveTurn, outcome: TurnOutcome) -> None:
        if not turn.done.done():
            turn.done.set_result(outcome)

    async def _answer_permission(self, pending: PendingPermission, allow: bool, by: str) -> None:
        connection = self._registry.get(pending.worker_id)
        if connection is not None:
            try:
                await connection.send(permission_reply_frame(pending.request_id, allow))
            except Exception:
                logger.warning("could not send a permission reply to worker %s", pending.worker_id)
        sender = "user" if by == "user" else "system"
        await self._post(
            pending.group_id, sender, "permission_result", {"request_id": pending.request_id, "allow": allow, "by": by}
        )

    async def _deny_expired(self, pending: PendingPermission) -> None:
        await self._answer_permission(pending, False, by="timeout")

    async def _post(self, group_id: int, sender: str, kind: str, body: dict) -> ForgeMessage:
        message = await self._store.append(group_id, sender, kind, body)
        self._bus.publish(group_id, message_event(message))
        return message

    def _publish_turn(self, group_id: int, running: bool) -> None:
        self._bus.publish(group_id, {"type": "turn", "running": running})

    def _groups(self, session: AsyncSession, account_id: int) -> GroupService:
        return GroupService(session, account_id, PlanPolicyRegistry(session))

    async def _owned_group(self, account_id: int, group_id: int) -> None:
        async with self._database.sessions() as session:
            await self._groups(session, account_id).get(group_id)
```

```python
    last_event_id = request.headers.get("last-event-id", "")

def _bearer(header: str | None) -> str | None:
    if header and header[:7].lower() == "bearer ":
        return header[7:].strip() or None
    return None


@router.websocket("/worker")
async def worker_socket(websocket: WebSocket) -> None:
    """A worker connects out with its pairing token, says hello (with its plan), then trades
    frames with the coordinator until it disconnects."""
    forge: ForgeRuntime = websocket.app.state.forge
    database = websocket.app.state.database
    token = _bearer(websocket.headers.get("authorization"))
    worker = None
    if token:
        async with database.sessions() as session:
            worker = await WorkerAuthenticator(session).authenticate(token)
    if worker is None:
        await websocket.close(code=4401)
        return
    await websocket.accept()
    try:
        hello = parse_worker_frame(await asyncio.wait_for(websocket.receive_text(), HELLO_TIMEOUT_SECONDS))
    except (asyncio.TimeoutError, WebSocketDisconnect):
        await websocket.close(code=4400)
        return
    if not isinstance(hello, Hello):
        await websocket.close(code=4400)
        return

    connection = WorkerConnection(worker.id, websocket)
    replaced = forge.registry.register(connection)
    if replaced is not None:
        await replaced.close(4000)
    try:
        async with database.sessions() as session:
            await WorkerAuthenticator(session).record_hello(worker.id, hello.plan)
        await connection.send(ready_frame(worker.id))
        while True:
            frame = parse_worker_frame(await websocket.receive_text())
            if frame is None:
                continue
            if isinstance(frame, Heartbeat):
                async with database.sessions() as session:
                    await WorkerAuthenticator(session).touch(worker.id)
                continue
            await forge.coordinator.on_worker_frame(worker.id, frame)
    except WebSocketDisconnect:
        pass
    finally:
        forge.registry.drop(worker.id, connection)
        await forge.coordinator.on_worker_disconnected(worker.id)
```

Registration order matters: FastAPI matches `/groups/{group_id}/messages` etc. by path, and the socket route is `/api/forge/worker` (prefix + `/worker`), so there is no clash.
