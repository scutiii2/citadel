```python
"""One-shot: move garrison modules into agents/, providers/, tools/, runtime/ and rewrite imports.

Usage (from apps/garrison):  python scripts/restructure_v2.py --check | --apply
Deleted after the move commit."""

from __future__ import annotations

import ast
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
SRC = ROOT / "src"

ORCHESTRATOR = ["delegation", "plan", "ask_user", "memory_recall", "skill_recall", "skill_reflection",
                "context_summarizer", "auto_route"]
ROUTER = ["router_client", "router_contract", "router_engine", "router_routing", "router_service"]
PROVIDERS = ["base_provider", "cancellation", "cooldown", "gateway_failover", "token_limits", "model_limits",
             "model_tiers", "reasoning_effort", "llm_config", "llm_options", "context_breakdown", "turn_guard"]
LANGUAGE = ["anthropic_provider", "openai_provider"]
DECISION = ["decision", "laya_model", "laya_provider", "laya_classifier", "jev_classifier",
            "typesafe_classifier", "triage_flow", "triage_strategy"]

# old module (dotted, under src) -> new module
MODMAP: dict[str, str] = {}


def _add(old_pkg: str, names: list[str], new_pkg: str) -> None:
    for name in names:
        MODMAP[f"src.{old_pkg}.{name}"] = f"src.{new_pkg}.{name}"


_add("agents", ORCHESTRATOR, "agents.roles.orchestrator")
_add("agents", ROUTER, "agents.roles.router")
_add("llm", ["agent_roles", "auto_tier"], "agents")
_add("llm", PROVIDERS, "providers")
_add("llm", LANGUAGE, "providers.language")
_add("llm", DECISION, "providers.decision")
for path in sorted((SRC / "core").glob("*.py")):
    if path.stem != "__init__":
        MODMAP[f"src.core.{path.stem}"] = f"src.runtime.{path.stem}"
for pkg in ("mcp_client", "private_extensions"):
    for path in sorted((SRC / pkg).glob("*.py")):
        if path.stem != "__init__":
            MODMAP[f"src.{pkg}.{path.stem}"] = f"src.tools.{pkg}.{path.stem}"

PACKAGE_MAP = {"src.mcp_client": "src.tools.mcp_client", "src.private_extensions": "src.tools.private_extensions"}
OLD_PACKAGES = {"src.llm", "src.core", "src.agents", "src.mcp_client", "src.private_extensions"}
_DOTTED = re.compile(r"\bsrc\.(?:llm|core|agents|mcp_client|private_extensions)(?:\.\w+)*")


def to_path(dotted: str) -> Path:
    return ROOT / (dotted.replace(".", "/") + ".py")


def new_module(old: str) -> str | None:
    """New dotted name for an old module or package; None when it does not move."""
    if old in MODMAP:
        return MODMAP[old]
    return PACKAGE_MAP.get(old)


def rewrite_dotted(text: str) -> str:
    """Rewrite every dotted reference, longest module prefix first (covers patch("src.llm.x.y") strings)."""

    def sub(match: re.Match[str]) -> str:
        parts = match.group(0).split(".")
        for end in range(len(parts), 1, -1):
            moved = new_module(".".join(parts[:end]))
            if moved:
                return ".".join([moved, *parts[end:]])
        return match.group(0)

    return _DOTTED.sub(sub, text)


def rewrite_imports(source: str) -> str:
    """Rewrite `from src.x import a, b` forms with the AST (names of one old package may split across new ones)."""
    tree = ast.parse(source)
    lines = source.splitlines(keepends=True)
    edits: list[tuple[int, int, str]] = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.ImportFrom) or node.level or not node.module:
            continue
        if node.module not in OLD_PACKAGES:
            continue
        groups: dict[str, list[str]] = {}
        for alias in node.names:
            target = new_module(f"{node.module}.{alias.name}")
            module = target.rsplit(".", 1)[0] if target and node.module not in PACKAGE_MAP else node.module
            if target and node.module in PACKAGE_MAP:
                module = target.rsplit(".", 1)[0]
            name = alias.name if alias.asname is None else f"{alias.name} as {alias.asname}"
            groups.setdefault(module, []).append(name)
        indent = " " * node.col_offset
        text = "".join(f"{indent}from {module} import {', '.join(names)}\n" for module, names in groups.items())
        edits.append((node.lineno - 1, node.end_lineno, text))
    for start, end, text in sorted(edits, reverse=True):
        lines[start:end] = [text]
    return rewrite_dotted("".join(lines))


def python_files() -> list[Path]:
    return [p for folder in ("src", "tests", "scripts") for p in (ROOT / folder).rglob("*.py")
            if "__pycache__" not in p.parts and p.name != "restructure_v2.py"]


def git(*args: str) -> None:
    subprocess.run(["git", *args], cwd=ROOT, check=True)


def main(apply: bool) -> int:
    missing = [old for old in MODMAP if not to_path(old).exists()]
    if missing:
        print("missing source files:", *missing, sep="\n  ")
        return 1
    for old, new in sorted(MODMAP.items()):
        print(f"{old} -> {new}")
    if not apply:
        return 0
    for old, new in MODMAP.items():
        target = to_path(new)
        target.parent.mkdir(parents=True, exist_ok=True)
        git("mv", str(to_path(old).relative_to(ROOT)), str(target.relative_to(ROOT)))
    for path in python_files():
        original = path.read_text(encoding="utf-8")
        updated = rewrite_imports(original)
        if updated != original:
            path.write_text(updated, encoding="utf-8")
    for new in set(MODMAP.values()):
        package = to_path(new).parent
        while package != SRC:
            init = package / "__init__.py"
            if not init.exists():
                init.write_text("", encoding="utf-8")
                git("add", str(init.relative_to(ROOT)))
            package = package.parent
    for pkg in ("llm", "core", "mcp_client", "private_extensions"):
        init = SRC / pkg / "__init__.py"
        if init.exists():
            git("rm", "-q", "-f", str(init.relative_to(ROOT)))
    return 0


if __name__ == "__main__":
    if len(sys.argv) != 2 or sys.argv[1] not in ("--check", "--apply"):
        sys.exit(__doc__)
    sys.exit(main(sys.argv[1] == "--apply"))
```

```python
"""Decision gateway `jev`: TypeSafe's Jev API (POST /v1/systemone), a cloud call.

Jev takes one `choice` question per call, so score levels and noul true/false are asked as a choice over
those options and mapped back to the Laya answer shape the rest of Garrison reads."""

from __future__ import annotations

import json
import os
import urllib.error
import urllib.request
from typing import Any, Callable, Mapping

from src.providers import llm_config
from src.providers.decision.contract import DecisionError, DecisionRequest, GatewayResult

ID = "jev"
KEY_ENV = "TYPESAFE_API_KEY"
_QUESTION_ID = "answer"
_NOUL_DEFAULTS = {"false": "No: the statement is false.", "true": "Yes: the statement is true."}
_DEFAULT_TIMEOUT = 3.0

Post = Callable[[str, dict[str, Any], str, float], dict[str, Any]]


def _post_json(url: str, payload: dict[str, Any], key: str, timeout: float) -> dict[str, Any]:
    request = urllib.request.Request(
        url, data=json.dumps(payload).encode("utf-8"), method="POST",
        headers={"Authorization": f"Bearer {key}", "Content-Type": "application/json"},
    )
    try:
        with urllib.request.urlopen(request, timeout=timeout) as response:
            return json.loads(response.read().decode("utf-8"))
    except urllib.error.HTTPError as error:
        raise DecisionError(f"TypeSafe answered HTTP {error.code}") from error
    except (urllib.error.URLError, TimeoutError, OSError) as error:
        raise DecisionError(f"TypeSafe could not be reached: {error}") from error
    except ValueError as error:
        raise DecisionError("TypeSafe sent something that is not JSON") from error


def _options(spec: Mapping[str, Any]) -> dict[str, str]:
    """The question's options as Jev names them: choices as given, score levels "0", "1", ... and noul "false"/"true"."""
    if spec["type"] == "choice":
        return dict(spec["criteria"])
    if spec["type"] == "score":
        return {str(i): f"level {i}: {level}" for i, level in enumerate(spec["criteria"])}
    return dict(spec.get("criteria") or _NOUL_DEFAULTS)


def _question_text(spec: Mapping[str, Any]) -> str:
    text = spec["instructions"]
    return text + " Pick the level that fits best; level 0 is the lowest." if spec["type"] == "score" else text


def _parse(body: Any, options: Mapping[str, str]) -> tuple[str, float, dict[str, float] | None, int, str]:
    try:
        answer = body["answers"][_QUESTION_ID]
        choice, confidence = answer["choice"], answer["confidence"]
        model = str(body.get("model", ""))
        tokens = body.get("usage", {}).get("input_tokens", 0)
    except (KeyError, TypeError, AttributeError) as error:
        raise DecisionError("TypeSafe's answer has an unexpected shape") from error
    if choice not in options:
        raise DecisionError(f"the answer names an option that was not offered: {choice!r}")
    if isinstance(confidence, bool) or not isinstance(confidence, (int, float)) or not 0 <= confidence <= 1:
        raise DecisionError("TypeSafe's answer has no usable confidence")
    if isinstance(tokens, bool) or not isinstance(tokens, int) or tokens < 0:
        tokens = 0
    raw = answer.get("probabilities")
    real = None
    if isinstance(raw, dict) and set(raw) == set(options) and all(
            not isinstance(v, bool) and isinstance(v, (int, float)) and 0 <= v <= 1 for v in raw.values()):
        real = {name: float(raw[name]) for name in options}
    return choice, float(confidence), real, tokens, model


def _answer(spec: Mapping[str, Any], choice: str, confidence: float, real: dict[str, float] | None,
            min_confidence: float) -> dict[str, Any]:
    """Jev's pick in the shape of a Laya answer. A reported distribution is kept; otherwise the other options share what is left."""
    out: dict[str, Any] = {"type": spec["type"]}
    if spec["type"] == "choice":
        others = [name for name in spec["criteria"] if name != choice]
        share = round((1 - confidence) / len(others), 4) if others else 0.0
        out["choice"] = choice
        out["probabilities"] = (
            {name: round(real[name], 4) for name in spec["criteria"]} if real
            else {name: (confidence if name == choice else share) for name in spec["criteria"]}
        )
    elif spec["type"] == "score":
        out["score"] = float(choice)
        out["legend"] = {str(i): level for i, level in enumerate(spec["criteria"])}
    else:
        out["noul"] = confidence if choice == "true" else round(1 - confidence, 4)
    out["answer_confidence"] = confidence
    out["uncertain"] = confidence < min_confidence
    return out


class JevGateway:
    id = ID
    local = False

    def __init__(self, post: Post | None = None) -> None:
        self._post = post or _post_json

    def _block(self) -> dict[str, Any] | None:
        try:
            block = llm_config.gateway("decision", ID)
        except (KeyError, ValueError):
            return None
        return block if str(block.get("api_key", "")).strip() else None

    def available(self) -> bool:
        return bool(os.environ.get(KEY_ENV, "").strip()) and self._block() is not None

    def ask(self, request: DecisionRequest, timeout: float | None = None) -> GatewayResult:
        block = self._block()
        key = os.environ.get(KEY_ENV, "").strip()
        if not key or block is None:
            raise DecisionError(f"{KEY_ENV} is not set")
        answers: dict[str, dict[str, Any]] = {}
        tokens, model = 0, str(block.get("model", ""))
        for qid, spec in request.questions.items():
            options = _options(spec)
            payload = {"state": request.text, "model": block["model"],
                       "questions": {_QUESTION_ID: {"type": "choice", "instructions": _question_text(spec),
                                                    "criteria": options}}}
            body = self._post(block["url"], payload, key, timeout or _DEFAULT_TIMEOUT)
            choice, confidence, real, used, reported = _parse(body, options)
            tokens += used
            model = reported or model
            answers[qid] = _answer(spec, choice, confidence, real, request.min_confidence)
        return GatewayResult(answers, tokens, model)
```

"""Decision gateway `laya`: the shared LayaModel service over HTTP. Garrison loads no model.

Which service strategy answers is gateway config (gateways/decision/laya.json `strategy`)."""

from __future__ import annotations

from typing import Any

from src.providers import llm_config
from src.providers.decision import laya_model
from src.providers.decision.contract import DecisionError, DecisionRequest, GatewayResult
from src.providers.decision.laya_model import LayaServiceError

ID = "laya"
WINDOW = 512
MAX_CHARS = 4000
DEFAULT_STRATEGY = "english-then-typed"


class LayaGateway:
    id = ID
    local = True

    def __init__(self, service: Any = None) -> None:
        self._service = service

    def _client(self) -> Any:
        return self._service or laya_model.agent()

    @property
    def strategy(self) -> str:
        block = llm_config.load_gateways().get("decision", {}).get(ID, {})
        return str(block.get("strategy") or DEFAULT_STRATEGY)

    def available(self) -> bool:
        return bool(self._client().configured)

    def ask(self, request: DecisionRequest, timeout: float | None = None) -> GatewayResult:
        if len(request.text) > MAX_CHARS:
            raise DecisionError(f"Laya takes at most {MAX_CHARS} characters of text")
        try:
            body = self._client().ask(self.strategy, request.text, request.questions, WINDOW,
                                      min_confidence=request.min_confidence)
        except LayaServiceError as error:
            raise DecisionError(str(error)) from error
        tokens = sum(call.get("input_tokens") or 0 for call in body.get("calls", []))
        return GatewayResult(body["answers"], tokens, self.strategy)

```python
"""Runs a decision strategy over the gateway arrangement [laya, jev].

laya-only / jev-only ask one gateway. `both` asks laya first and jev only about the questions laya failed
to settle (error, or confidence below the request's minimum). The agent wrapper picks the strategy; gateways
never chain among themselves."""

from __future__ import annotations

import time
from dataclasses import dataclass, replace
from datetime import datetime, timezone
from typing import Any, Callable, Mapping

import anyio.to_thread

from src.providers.decision.contract import DecisionError, DecisionGateway, DecisionRequest

STRATEGIES = ("laya-only", "jev-only", "both")
ARRANGEMENT = ("laya", "jev")
_ORDER = {"laya-only": ("laya",), "jev-only": ("jev",), "both": ARRANGEMENT}
_EPSILON = 1e-9


@dataclass(frozen=True)
class DecisionResult:
    answers: dict[str, dict[str, Any]]
    tokens: int
    unsettled: tuple[str, ...]


@dataclass(frozen=True)
class Choice:
    choice: str
    confidence: int
    prompt_tokens: int
    backend: str
    model: str
    probabilities: dict[str, float] | None = None


def default_gateways() -> dict[str, DecisionGateway]:
    from src.providers.decision.gateways.jev import JevGateway
    from src.providers.decision.gateways.laya import LayaGateway

    return {"laya": LayaGateway(), "jev": JevGateway()}


def _settled(answer: Mapping[str, Any] | None, minimum: float) -> bool:
    return (answer is not None and not answer.get("uncertain", True) and not answer.get("unknown")
            and answer.get("answer_confidence", 0) >= minimum - _EPSILON)


def run(strategy: str, request: DecisionRequest, *, gateways: Mapping[str, DecisionGateway] | None = None,
        on_usage: Callable[[dict[str, Any]], None] | None = None,
        check: Callable[[], float] | None = None) -> DecisionResult:
    if strategy not in STRATEGIES:
        raise ValueError(f"unknown decision strategy {strategy!r}; choose one of: {', '.join(STRATEGIES)}")
    gateways = gateways if gateways is not None else default_gateways()
    names = [n for n in _ORDER[strategy] if not (request.local_only and not gateways[n].local)]
    pending = dict(request.questions)
    answers: dict[str, dict[str, Any]] = {}
    tokens = 0
    problems: list[str] = []
    for name in names:
        if not pending:
            break
        gateway = gateways[name]
        if not gateway.available():
            problems.append(f"{name} is not available")
            continue
        limit = check() if check else None
        started, started_at = time.monotonic(), datetime.now(timezone.utc).isoformat()
        result = None
        try:
            result = gateway.ask(replace(request, questions=pending), limit)
        except DecisionError as error:
            problems.append(f"{name}: {error}")
        finally:
            if on_usage:
                on_usage({"provider_id": name, "model": result.model if result else name, "started_at": started_at,
                          "duration_ms": round((time.monotonic() - started) * 1000),
                          "input_tokens": result.tokens if result else None,
                          "output_tokens": 0 if result else None, "ok": result is not None,
                          "reason": "" if result else "backend_failed"})
        if result is None:
            continue
        tokens += result.tokens
        for qid, answer in result.answers.items():
            answers[qid] = {**answer, "answered_by": name}
        pending = {q: s for q, s in pending.items() if not _settled(answers.get(q), request.min_confidence)}
    if not answers:
        raise DecisionError("; ".join(problems) or "no decision gateway could be asked")
    for qid in pending:
        if qid in answers:
            answers[qid]["uncertain"] = True
    return DecisionResult(answers, tokens, tuple(q for q in pending if q in answers))


async def arun(strategy: str, request: DecisionRequest, **kwargs: Any) -> DecisionResult:
    return await anyio.to_thread.run_sync(lambda: run(strategy, request, **kwargs))


def ask_choice(subject: str, criteria: Mapping[str, str], question: str, *, strategy: str, min_confidence: int,
               gateways: Mapping[str, DecisionGateway] | None = None,
               on_usage: Callable[[dict[str, Any]], None] | None = None, timeout: float | None = None) -> Choice:
    """One choice question: the answer with its confidence in percent (0-100)."""
    if len(criteria) < 2:
        raise DecisionError("a choice needs at least two options")
    request = DecisionRequest(subject, {"answer": {"type": "choice", "instructions": question, "criteria": dict(criteria)}},
                              min_confidence=min_confidence / 100)
    result = run(strategy, request, gateways=gateways, on_usage=on_usage,
                 check=(lambda: timeout) if timeout else None)
    answer = result.answers["answer"]
    return Choice(answer["choice"], round(answer["answer_confidence"] * 100), result.tokens, answer["answered_by"],
                  answer["answered_by"], answer.get("probabilities"))
```

def check_laya_strategy(uses_laya: bool) -> None:
