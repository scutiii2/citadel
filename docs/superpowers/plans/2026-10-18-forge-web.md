Create `src/services/forgeStream.ts`:

```ts
import { forgeClient, type ForgeEvent } from "../api/ForgeClient";
import { reportUnauthorized, UnauthorizedError } from "../api/http";

/** How a watch ended. "gone": the group no longer exists. "aborted": the caller stopped it. */
export type ForgeWatchEnd = "gone" | "aborted";

const RECONNECT_DELAYS_MS = [500, 1000, 2000, 4000, 8000];

/** Splits whole server-sent events off the front of `buffer`. A partial event stays in `rest`;
 * a keep-alive comment or a damaged block yields nothing. */
export function takeEvents(buffer: string): { events: ForgeEvent[]; rest: string } {
  const events: ForgeEvent[] = [];
  let rest = buffer;
  let boundary: number;
  while ((boundary = rest.indexOf("\n\n")) !== -1) {
    const block = rest.slice(0, boundary);
    rest = rest.slice(boundary + 2);
    const data = block
      .split("\n")
      .filter((line) => line.startsWith("data: "))
      .map((line) => line.slice(6))
      .join("\n");
    if (!data) continue;
    try {
      events.push(JSON.parse(data) as ForgeEvent);
    } catch {
      /* skip a damaged block */
    }
  }
  return { events, rest };
}

/** Follows a group's event stream until the group is gone or `signal` aborts. fetch() rather
 * than EventSource: it can be aborted, sees the 401/404 status, and resumes from the last
 * message `seq` after a dropped connection. */
export async function watchGroup(
  groupId: number,
  after: number,
  onEvent: (event: ForgeEvent) => void,
  signal: AbortSignal,
): Promise<ForgeWatchEnd> {
  let cursor = after;
  for (let attempt = 0; ; attempt += 1) {
    try {
      const outcome = await readStream(groupId, cursor, signal, (event) => {
        if (event.type === "message") cursor = Math.max(cursor, event.seq);
        attempt = 0;
        onEvent(event);
      });
      if (outcome === "gone") return "gone";
    } catch (err) {
      if (signal.aborted) return "aborted";
      if (err instanceof UnauthorizedError) throw err;
    }
    if (signal.aborted) return "aborted";
    const delay = RECONNECT_DELAYS_MS[attempt];
    if (delay === undefined) throw new Error("Lost the connection to ember_api");
    await sleep(delay, signal);
    if (signal.aborted) return "aborted";
  }
}

async function readStream(
  groupId: number,
  after: number,
  signal: AbortSignal,
  onEvent: (event: ForgeEvent) => void,
): Promise<"gone" | "dropped"> {
  const response = await fetch(forgeClient.eventsUrl(groupId, after), {
    credentials: "same-origin",
    headers: { Accept: "text/event-stream" },
    signal,
  });
  if (response.status === 404) return "gone";
  if (response.status === 401) {
    reportUnauthorized();
    throw new UnauthorizedError(401, "Not logged in");
  }
  if (!response.ok || !response.body) return "dropped";

  const reader = response.body.pipeThrough(new TextDecoderStream()).getReader();
  let buffer = "";
  for (;;) {
    const { value, done } = await reader.read();
    if (done) return "dropped";
    const taken = takeEvents(buffer + value);
    buffer = taken.rest;
    for (const event of taken.events) {
      onEvent(event);
      if (event.type === "lagged") {
        await reader.cancel();
        return "dropped";
      }
    }
  }
}

function sleep(ms: number, signal: AbortSignal): Promise<void> {
  return new Promise((resolve) => {
    const timer = setTimeout(resolve, ms);
    signal.addEventListener("abort", () => {
      clearTimeout(timer);
      resolve();
    });
  });
}
```

```ts
import type { ForgeDeltaEvent, ForgeEvent, ForgeMessageEvent } from "../api/ForgeClient";

/** What the transcript shows, built from the event stream. Pure: every function returns new values. */

export type PermissionStatus = "pending" | "allowed" | "denied" | "timed_out";

export interface UserItem {
  key: string;
  kind: "user";
  seq: number;
  text: string;
  time: string;
}
export interface WorkerItem {
  key: string;
  kind: "worker";
  /** Null while it is a streaming draft. */
  seq: number | null;
  turnId: string;
  label: string;
  provider: string | null;
  text: string;
  streaming: boolean;
  /** A draft whose turn ended without a stored reply. */
  partial: boolean;
  time: string | null;
  usage: { input_tokens: number; output_tokens: number } | null;
}
export interface ToolItem {
  key: string;
  kind: "tool";
  seq: number;
  label: string;
  name: string;
  summary: string;
}
export interface PermissionItem {
  key: string;
  kind: "permission";
  seq: number;
  requestId: string;
  turnId: string;
  label: string;
  tool: string;
  summary: string;
  status: PermissionStatus;
}
export interface SystemItem {
  key: string;
  kind: "system";
  seq: number;
  text: string;
}
export type TranscriptItem = UserItem | WorkerItem | ToolItem | PermissionItem | SystemItem;

export interface TranscriptState {
  items: TranscriptItem[];
  /** The newest stored message seen. */
  cursor: number;
  running: boolean;
}

export const emptyTranscript = (): TranscriptState => ({ items: [], cursor: 0, running: false });

const text = (value: unknown): string => (typeof value === "string" ? value : "");

export function applyEvent(state: TranscriptState, event: ForgeEvent): TranscriptState {
  switch (event.type) {
    case "turn":
      if (state.running === event.running) return state;
      return { ...state, running: event.running, items: event.running ? state.items : settleDrafts(state.items) };
    case "delta":
      return appendDelta(state, event);
    case "message":
      return event.seq <= state.cursor ? state : addMessage({ ...state, cursor: event.seq }, event);
    default:
      return state;
  }
}

function settleDrafts(items: TranscriptItem[]): TranscriptItem[] {
  if (!items.some((item) => item.kind === "worker" && item.streaming)) return items;
  return items.map((item) => (item.kind === "worker" && item.streaming ? { ...item, streaming: false, partial: true } : item));
}

function draftIndex(items: TranscriptItem[], turnId: string): number {
  return items.findIndex((item) => item.kind === "worker" && item.seq === null && item.turnId === turnId);
}

function appendDelta(state: TranscriptState, event: ForgeDeltaEvent): TranscriptState {
  const at = draftIndex(state.items, event.turn_id);
  if (at >= 0) {
    const draft = state.items[at] as WorkerItem;
    const items = state.items.slice();
    items[at] = { ...draft, text: draft.text + event.text };
    return { ...state, items };
  }
  const draft: WorkerItem = {
    key: `d${event.turn_id}`,
    kind: "worker",
    seq: null,
    turnId: event.turn_id,
    label: event.label,
    provider: null,
    text: event.text,
    streaming: true,
    partial: false,
    time: null,
    usage: null,
  };
  return { ...state, items: [...state.items, draft] };
}

function addMessage(state: TranscriptState, event: ForgeMessageEvent): TranscriptState {
  const body = event.body;
  const key = `m${event.seq}`;
  switch (event.kind) {
    case "text":
      return event.sender === "user"
        ? push(state, { key, kind: "user", seq: event.seq, text: text(body.text), time: event.created_at })
        : withReply(state, event);
    case "tool":
      return push(state, { key, kind: "tool", seq: event.seq, label: text(body.label), name: text(body.name), summary: text(body.summary) });
    case "permission_request":
      return push(state, {
        key, kind: "permission", seq: event.seq, requestId: text(body.request_id), turnId: text(body.turn_id),
        label: text(body.label), tool: text(body.tool), summary: text(body.summary), status: "pending",
      });
    case "permission_result":
      return resolvePermission(state, text(body.request_id), body.allow === true, text(body.by));
    case "system":
      return push(state, { key, kind: "system", seq: event.seq, text: text(body.text) });
    default:
      return state;
  }
}

function push(state: TranscriptState, item: TranscriptItem): TranscriptState {
  return { ...state, items: [...state.items, item] };
}

function withReply(state: TranscriptState, event: ForgeMessageEvent): TranscriptState {
  const body = event.body;
  const usage = body.usage as { input_tokens?: number; output_tokens?: number } | undefined;
  const reply: WorkerItem = {
    key: `m${event.seq}`,
    kind: "worker",
    seq: event.seq,
    turnId: text(body.turn_id),
    label: text(body.label),
    provider: text(body.provider) || null,
    text: text(body.text),
    streaming: false,
    partial: false,
    time: event.created_at,
    usage: usage ? { input_tokens: usage.input_tokens ?? 0, output_tokens: usage.output_tokens ?? 0 } : null,
  };
  const at = draftIndex(state.items, reply.turnId);
  if (at < 0) return push(state, reply);
  const items = state.items.slice();
  items[at] = reply;
  return { ...state, items };
}

function resolvePermission(state: TranscriptState, requestId: string, allow: boolean, by: string): TranscriptState {
  const at = state.items.findIndex((item) => item.kind === "permission" && item.requestId === requestId);
  if (at < 0) return state;
  const status: PermissionStatus = by === "timeout" ? "timed_out" : allow ? "allowed" : "denied";
  const items = state.items.slice();
  items[at] = { ...(state.items[at] as PermissionItem), status };
  return { ...state, items };
}
```

```ts
/** Helpers for the composer's @mention menu. */

export interface MentionQuery {
  /** Index of the "@". */
  start: number;
  /** What was typed after it. */
  query: string;
}

const TYPING = /(?:^|\s)@([A-Za-z0-9_-]{0,24})$/;

/** The @mention being typed right before `caret`, or null. */
export function mentionAt(text: string, caret: number): MentionQuery | null {
  const match = TYPING.exec(text.slice(0, caret));
  if (!match) return null;
  const query = match[1]!;
  return { start: caret - query.length - 1, query };
}

export function matchingLabels(labels: string[], query: string): string[] {
  const wanted = query.toLowerCase();
  return labels.filter((label) => label.toLowerCase().startsWith(wanted));
}

/** Writes `@label ` in place of the typed `@query`; the caret lands after the space. */
export function completeMention(
  text: string,
  mention: MentionQuery,
  caret: number,
  label: string,
): { text: string; caret: number } {
  const inserted = `@${label} `;
  return { text: text.slice(0, mention.start) + inserted + text.slice(caret), caret: mention.start + inserted.length };
}
```

```ts
import { defineStore } from "pinia";
import { computed, ref, watch } from "vue";
import {
  forgeClient,
  type ForgeCreatedWorker,
  type ForgeGroup,
  type ForgeGroupPatch,
  type ForgePlans,
  type ForgeProvider,
  type ForgeWorker,
} from "../api/ForgeClient";
import { watchGroup } from "../services/forgeStream";
import { errorMessage } from "../utils/errors";
import { applyEvent, emptyTranscript, type TranscriptState } from "../utils/forgeTranscript";
import { useAuthStore } from "./auth";

type CreateMode = Parameters<typeof forgeClient.createGroup>[1];

/** The account's Forge workers and groups, and the live transcript of the open group.
 * Loaded on first use and dropped when the user changes. Actions that change something throw
 * ember_api's message so the dialog that called them can show it. */
export const useForgeStore = defineStore("forge", () => {
  const auth = useAuthStore();

  const workers = ref<ForgeWorker[]>([]);
  const plans = ref<ForgePlans>({ claude: [], codex: [] });
  const groups = ref<ForgeGroup[]>([]);
  const selectedId = ref<number | null>(null);
  const transcript = ref<TranscriptState>(emptyTranscript());
  const connection = ref<"idle" | "live" | "lost">("idle");
  const loading = ref(false);
  const loadError = ref("");
  let loaded = false;
  let controller: AbortController | null = null;
  // Bumped on every account change: answers and events for the previous account are ignored.
  let generation = 0;

  const allowed = computed(() => auth.hasPermission("forge.use"));
  const selected = computed(() => groups.value.find((g) => g.id === selectedId.value) ?? null);

  watch(
    () => (allowed.value ? (auth.account?.id ?? null) : null),
    () => {
      generation += 1;
      controller?.abort();
      controller = null;
      workers.value = [];
      groups.value = [];
      plans.value = { claude: [], codex: [] };
      selectedId.value = null;
      transcript.value = emptyTranscript();
      connection.value = "idle";
      loading.value = false;
      loadError.value = "";
      loaded = false;
    },
  );

  async function refresh(): Promise<void> {
    if (!allowed.value) return;
    const started = generation;
    if (!loaded) loading.value = true;
    try {
      const [nextWorkers, nextGroups, nextPlans] = await Promise.all([
        forgeClient.workers(),
        forgeClient.groups(),
        forgeClient.plans(),
      ]);
      if (started !== generation) return;
      workers.value = nextWorkers;
      groups.value = nextGroups;
      plans.value = nextPlans;
      loadError.value = "";
      loaded = true;
      if (selectedId.value !== null && !nextGroups.some((g) => g.id === selectedId.value)) select(null);
    } catch (err) {
      if (started === generation) loadError.value = errorMessage(err);
    } finally {
      if (started === generation) loading.value = false;
    }
  }

  async function ensureLoaded(): Promise<void> {
    if (!loaded) await refresh();
  }

  function startWatch(id: number, after: number): void {
    const mine = new AbortController();
    controller = mine;
    const started = generation;
    connection.value = "live";
    watchGroup(
      id,
      after,
      (event) => {
        if (!mine.signal.aborted && started === generation) transcript.value = applyEvent(transcript.value, event);
      },
      mine.signal,
    )
      .then(async (end) => {
        if (end === "gone" && !mine.signal.aborted) await refresh();
      })
      .catch(() => {
        if (!mine.signal.aborted && started === generation) connection.value = "lost";
      });
  }

  function select(id: number | null): void {
    controller?.abort();
    controller = null;
    selectedId.value = id;
    transcript.value = emptyTranscript();
    connection.value = "idle";
    if (id !== null) startWatch(id, 0);
  }

  /** Watches the open group again, from the newest message already shown. */
  function reconnect(): void {
    const id = selectedId.value;
    if (id === null) return;
    controller?.abort();
    startWatch(id, transcript.value.cursor);
  }

  /** Stops the live watch (the page was left) but remembers which group was open. */
  function leave(): void {
    controller?.abort();
    controller = null;
    connection.value = "idle";
  }

  function upsert(group: ForgeGroup): void {
    const at = groups.value.findIndex((g) => g.id === group.id);
    if (at < 0) groups.value = [group, ...groups.value];
    else groups.value.splice(at, 1, group);
  }

  async function createGroup(title: string, mode: CreateMode): Promise<ForgeGroup> {
    const group = await forgeClient.createGroup(title, mode);
    upsert(group);
    select(group.id);
    return group;
  }

  async function updateGroup(id: number, patch: ForgeGroupPatch): Promise<ForgeGroup> {
    const group = await forgeClient.updateGroup(id, patch);
    upsert(group);
    return group;
  }

  async function removeGroup(id: number): Promise<void> {
    await forgeClient.removeGroup(id);
    groups.value = groups.value.filter((g) => g.id !== id);
    if (selectedId.value === id) select(null);
  }

  async function invite(groupId: number, workerId: number, folder: string): Promise<void> {
    upsert(await forgeClient.invite(groupId, workerId, folder));
  }

  async function removeMember(groupId: number, workerId: number): Promise<void> {
    upsert(await forgeClient.removeMember(groupId, workerId));
  }

  /** The token is returned for the caller to show once; the store keeps only the worker. */
  async function createWorker(provider: ForgeProvider, label: string): Promise<ForgeCreatedWorker> {
    const created = await forgeClient.createWorker(provider, label);
    const { token: _token, ...worker } = created;
    workers.value = [...workers.value, worker];
    return created;
  }

  async function removeWorker(id: number): Promise<void> {
    await forgeClient.removeWorker(id);
    await refresh();
  }

  async function savePlans(provider: ForgeProvider, list: string[]): Promise<void> {
    await forgeClient.setPlans(provider, list);
    await refresh();
  }

  async function send(text: string): Promise<void> {
    if (selectedId.value !== null) await forgeClient.send(selectedId.value, text);
  }

  async function stop(): Promise<void> {
    if (selectedId.value !== null) await forgeClient.stop(selectedId.value);
  }

  async function answer(requestId: string, allow: boolean): Promise<void> {
    if (selectedId.value !== null) await forgeClient.answerPermission(selectedId.value, requestId, allow);
  }

  return {
    workers, plans, groups, selectedId, selected, transcript, connection, loading, loadError,
    ensureLoaded, refresh, select, reconnect, leave,
    createGroup, updateGroup, removeGroup, invite, removeMember,
    createWorker, removeWorker, savePlans, send, stop, answer,
  };
});
```

```vue
<script setup lang="ts">
import { computed, nextTick, ref, watch } from "vue";
import { FORGE_MESSAGE_MAX } from "../api/ForgeClient";
import { completeMention, matchingLabels, mentionAt } from "../utils/forgeMention";

const props = defineProps<{
  /** The @handles of the group's workers. */
  labels: string[];
  running: boolean;
  busy: boolean;
  /** Non-empty: why nothing can be sent (no worker invited yet). */
  disabledReason: string;
  error: string;
}>();
const text = defineModel<string>({ required: true });
const emit = defineEmits<{ send: []; stop: [] }>();

const field = ref<HTMLTextAreaElement | null>(null);
const caret = ref(0);
const active = ref(0);
const dismissed = ref(false);

const mention = computed(() => mentionAt(text.value, caret.value));
const options = computed(() => (mention.value ? matchingLabels(props.labels, mention.value.query) : []));
const menuOpen = computed(() => options.value.length > 0 && !dismissed.value);
const blank = computed(() => text.value.trim() === "");
const canSend = computed(() => !blank.value && !props.busy && !props.running && props.disabledReason === "");
const nearLimit = computed(() => text.value.length > FORGE_MESSAGE_MAX * 0.9);

watch(text, () => {
  dismissed.value = false;
  active.value = 0;
});

function track(): void {
  caret.value = field.value?.selectionStart ?? text.value.length;
}

function pick(label: string): void {
  const current = mention.value;
  if (!current) return;
  const done = completeMention(text.value, current, caret.value, label);
  text.value = done.text;
  void nextTick(() => {
    field.value?.focus();
    field.value?.setSelectionRange(done.caret, done.caret);
    caret.value = done.caret;
  });
}

function onKey(event: KeyboardEvent): void {
  if (menuOpen.value) {
    const count = options.value.length;
    if (event.key === "ArrowDown") {
      event.preventDefault();
      active.value = (active.value + 1) % count;
      return;
    }
    if (event.key === "ArrowUp") {
      event.preventDefault();
      active.value = (active.value - 1 + count) % count;
      return;
    }
    if (event.key === "Enter" || event.key === "Tab") {
      event.preventDefault();
      pick(options.value[active.value]!);
      return;
    }
    if (event.key === "Escape") {
      event.preventDefault();
      dismissed.value = true;
      return;
    }
  }
  if (event.key === "Enter" && !event.shiftKey && !event.isComposing) {
    event.preventDefault();
    if (canSend.value) emit("send");
  }
}
</script>

<template>
  <form class="composer" @submit.prevent="canSend && emit('send')">
    <p v-if="disabledReason" class="reason">{{ disabledReason }}</p>
    <p v-if="error" class="error" role="alert">{{ error }}</p>
    <div class="row">
      <div class="field">
        <ul v-if="menuOpen" class="menu" role="listbox" aria-label="Workers">
          <li
            v-for="(label, index) in options"
            :key="label"
            role="option"
            :aria-selected="index === active"
            :class="{ active: index === active }"
            @mousedown.prevent="pick(label)"
          >
            @{{ label }}
          </li>
        </ul>
        <textarea
          ref="field"
          v-model="text"
          rows="2"
          :maxlength="FORGE_MESSAGE_MAX"
          :disabled="disabledReason !== ''"
          aria-label="Message"
          placeholder="Write to a worker with @name. Enter sends, Shift+Enter adds a line."
          @keydown="onKey"
          @input="track"
          @keyup="track"
          @click="track"
        />
        <span v-if="nearLimit" class="count">{{ text.length }}/{{ FORGE_MESSAGE_MAX }}</span>
      </div>
      <button v-if="running" type="button" class="stop" @click="emit('stop')">Stop</button>
      <button v-else type="submit" class="send" :disabled="!canSend" :aria-busy="busy">Send</button>
    </div>
  </form>
</template>
```

```vue
  () => {
    draft.value = "";
    sendError.value = "";
    panelError.value = "";
  },

function open(next: Dialog): void {
  dialogError.value = "";
  dialog.value = next;
}
function close(): void {
  if (dialogBusy.value) return;
  dialog.value = "";
  workerToRemove.value = null;
}

/** Runs a dialog's action: busy while it runs, closed when it succeeds, its error shown when it fails. */
async function runInDialog(action: () => Promise<void>, closeAfter = true): Promise<void> {
  dialogBusy.value = true;
  dialogError.value = "";
  try {
    await action();
    if (closeAfter) dialog.value = "";
  } catch (err) {
    dialogError.value = errorMessage(err);
  } finally {
    dialogBusy.value = false;
  }
}

/** Runs a change made from the side panel: its error shows under the panel, the page stays usable. */
async function runInPanel(action: () => Promise<unknown>): Promise<void> {
  panelBusy.value = true;
  panelError.value = "";
  try {
    await action();
  } catch (err) {
    panelError.value = errorMessage(err);
  } finally {
    panelBusy.value = false;
  }
}

async function send(): Promise<void> {
  if (sending.value) return;
  sending.value = true;
  sendError.value = "";
  try {
    await forge.send(draft.value.trim());
    draft.value = "";
  } catch (err) {
    sendError.value = errorMessage(err);
  } finally {
    sending.value = false;
  }
}

async function stop(): Promise<void> {
  try {
    await forge.stop();
  } catch (err) {
    sendError.value = errorMessage(err);
  }
}

async function answer(requestId: string, allow: boolean): Promise<void> {
  try {
    await forge.answer(requestId, allow);
  } catch (err) {
    sendError.value = errorMessage(err);
  }
}

const setMode = (mode: ForgeMode) => runInPanel(() => forge.updateGroup(group.value!.id, { mode }));
const setRelay = (relay: boolean) => runInPanel(() => forge.updateGroup(group.value!.id, { relay }));
const removeMember = (workerId: number) => runInPanel(() => forge.removeMember(group.value!.id, workerId));

const createGroup = (title: string, mode: Exclude<ForgeMode, "full_auto">) => runInDialog(() => forge.createGroup(title, mode).then(() => undefined));
const invite = (workerId: number, folder: string) => runInDialog(() => forge.invite(group.value!.id, workerId, folder));
const confirmFullAuto = () =>
  runInDialog(() => forge.updateGroup(group.value!.id, { mode: "full_auto", confirm_full_auto: true }).then(() => undefined));
const deleteGroup = () => runInDialog(() => forge.removeGroup(group.value!.id));

async function createWorker(provider: ForgeProvider, label: string): Promise<void> {
  await runInDialog(async () => {
    createdWorker.value = await forge.createWorker(provider, label);
  }, false);
}

function askRemoveWorker(worker: ForgeWorker): void {
  workerToRemove.value = worker;
  open("remove-worker");
}

async function removeWorker(): Promise<void> {
  const worker = workerToRemove.value;
  if (!worker) return;
  await runInDialog(async () => {
    await forge.removeWorker(worker.id);
    workerToRemove.value = null;
  });
  if (!dialogError.value) open("workers");
}

const savePlans = (provider: ForgeProvider, plans: string[]) => runInDialog(() => forge.savePlans(provider, plans), false);

function dismissToken(): void {
  createdWorker.value = null;
}
```
