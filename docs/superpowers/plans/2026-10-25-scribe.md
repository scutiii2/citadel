# Scribe (speech to text) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A standalone Scribe service transcribes an attached English audio file to text through one MCP tool, reached via Armory, with an upload path from ember_web through ember_api.

**Architecture:** `Python/Scribe/` is a FastMCP + Starlette service (port 8091) modelled on `Python/Bard/`. ember_api accepts base64 audio at `POST /api/attachments/audio`, forwards the raw bytes to Scribe's `POST /api/files`, and returns a `file_id`. ember_web puts an `[Audio file_id: …]` header line in the question; the agent calls `scribe__tool_scribe_transcribe`.

**Tech Stack:** Python >= 3.11, mcp==1.28.0, pydantic==2.13.4, uvicorn==0.49.0, python-dotenv==1.2.2, faster-whisper (pin in Task 1), ffmpeg on PATH, pytest + pytest-asyncio; ember_api (FastAPI); ember_web (Vue 3, vitest).

**Spec:** `Python/Citadel/docs/superpowers/specs/2026-10-25-scribe-design.md`. Bard is the template: `Python/Bard/src/*.py`, `Python/Bard/AGENTS.md`.

```python
    quota_bytes: int = 200 * 1024 * 1024

    timeout_s: int = 300
```

```python
"""Per-owner audio uploads with a JSON sidecar, a transcript cache, a time-to-live and a quota.

Blocking by design (plain file IO); the service calls it through asyncio.to_thread.
"""

from __future__ import annotations

import hashlib
import json
import os
import re
import time
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from src.errors import QuotaExceeded

_ID = re.compile(r"^[0-9a-f]{32}$")


@dataclass(frozen=True)
class StoredAudio:
    file_id: str
    path: Path
    filename: str
    duration_s: float
    size_bytes: int
    created_at: float


class AudioStore:
    def __init__(
        self, root: Path, ttl_seconds: int, quota_bytes: int, clock: Callable[[], float] = time.time
    ) -> None:
        self._root = Path(root)
        self._ttl = ttl_seconds
        self._quota = quota_bytes
        self._clock = clock

    @staticmethod
    def file_id(owner: str, data: bytes) -> str:
        return hashlib.sha256(owner.encode("utf-8") + b"\0" + data).hexdigest()[:32]

    def expires_at(self, stored: StoredAudio) -> float:
        return stored.created_at + self._ttl

    def _owner_dir(self, owner: str) -> Path:
        return self._root / hashlib.sha256(owner.encode("utf-8")).hexdigest()[:16]

    def _expired(self, stored: StoredAudio) -> bool:
        return self.expires_at(stored) <= self._clock()

    @staticmethod
    def _read(owner_dir: Path, file_id: str) -> StoredAudio | None:
        binary = owner_dir / f"{file_id}.bin"
        try:
            info = json.loads((owner_dir / f"{file_id}.json").read_text(encoding="utf-8"))
            return StoredAudio(
                file_id, binary, info["filename"], float(info["duration_s"]), binary.stat().st_size,
                float(info["created_at"]),
            )
        except (OSError, ValueError, KeyError):
            return None

    @staticmethod
    def _remove(owner_dir: Path, file_id: str) -> None:
        for suffix in (".bin", ".json", ".t.json"):
            (owner_dir / f"{file_id}{suffix}").unlink(missing_ok=True)

    @staticmethod
    def _write(path: Path, data: bytes) -> None:
        tmp = path.with_name(path.name + ".tmp")
        tmp.write_bytes(data)
        os.replace(tmp, path)

    def get(self, owner: str, file_id: str) -> StoredAudio | None:
        if not _ID.match(file_id):
            return None
        owner_dir = self._owner_dir(owner)
        stored = self._read(owner_dir, file_id)
        if stored is None:
            return None
        if self._expired(stored):
            self._remove(owner_dir, file_id)
            return None
        return stored

    def _sweep_dir(self, owner_dir: Path) -> int:
        removed = 0
        for sidecar in owner_dir.glob("*.json"):
            if not _ID.match(sidecar.stem):  # skips <id>.t.json transcripts
                continue
            stored = self._read(owner_dir, sidecar.stem)
            if stored is None or self._expired(stored):
                self._remove(owner_dir, sidecar.stem)
                removed += 1
        return removed

    def sweep(self) -> int:
        if not self._root.exists():
            return 0
        return sum(self._sweep_dir(d) for d in self._root.iterdir() if d.is_dir())

    def put(self, owner: str, file_id: str, data: bytes, filename: str, duration_s: float) -> StoredAudio:
        owner_dir = self._owner_dir(owner)
        owner_dir.mkdir(parents=True, exist_ok=True)
        self._sweep_dir(owner_dir)
        used = sum(p.stat().st_size for p in owner_dir.glob("*.bin"))
        if used + len(data) > self._quota:
            raise QuotaExceeded("Your audio storage is full. Wait for older audio to expire and try again.")
        created = self._clock()
        meta = {"filename": filename, "duration_s": duration_s, "created_at": created}
        self._write(owner_dir / f"{file_id}.json", json.dumps(meta).encode("utf-8"))
        self._write(owner_dir / f"{file_id}.bin", data)
        return StoredAudio(file_id, owner_dir / f"{file_id}.bin", filename, duration_s, len(data), created)

    def get_transcript(self, owner: str, file_id: str, model: str) -> dict | None:
        if not _ID.match(file_id):
            return None
        try:
            payload = json.loads((self._owner_dir(owner) / f"{file_id}.t.json").read_text(encoding="utf-8"))
        except (OSError, ValueError):
            return None
        return payload if payload.get("model") == model else None

    def put_transcript(self, owner: str, file_id: str, model: str, payload: dict) -> None:
        path = self._owner_dir(owner) / f"{file_id}.t.json"
        self._write(path, json.dumps({**payload, "model": model}).encode("utf-8"))
```

```python
from __future__ import annotations

from pydantic import BaseModel


class SegmentOut(BaseModel):
    start_s: float
    end_s: float
    text: str


class TranscribeResult(BaseModel):
    text: str
    duration_s: float
```

```python
"""Speech-to-text engines. ffmpeg decodes (async subprocess); the model runs in a worker thread."""

from __future__ import annotations

import asyncio
from dataclasses import dataclass
from pathlib import Path
from typing import Protocol

from src.errors import AudioRejected, EngineFailed

PCM_BYTES_PER_SECOND = 16000 * 2  # 16 kHz, mono, 16-bit


@dataclass(frozen=True)
class Segment:
    start_s: float
    end_s: float
    text: str


@dataclass(frozen=True)
class Transcript:
    text: str
    segments: list[Segment]
    language: str


class SttEngine(Protocol):
    model_name: str

    async def decode(self, path: Path) -> bytes:
        """16 kHz mono s16le PCM. Raises AudioRejected when ffmpeg cannot read the file."""

    async def transcribe(self, pcm: bytes) -> Transcript:
        """Raises EngineFailed."""


class WhisperEngine:
    def __init__(self, models_dir: Path, model_name: str, ffmpeg: str) -> None:
        self.model_name = model_name
        self._models_dir = Path(models_dir)
        self._ffmpeg = ffmpeg
        self._model = None
        self._lock = asyncio.Lock()

    async def decode(self, path: Path) -> bytes:
        try:
            process = await asyncio.create_subprocess_exec(
                self._ffmpeg, "-v", "error", "-nostdin", "-i", str(path),
                "-f", "s16le", "-ar", "16000", "-ac", "1", "pipe:1",
                stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
            )
        except OSError as error:
            raise EngineFailed("Audio decoding is unavailable.") from error
        pcm, _ = await process.communicate()
        if process.returncode != 0 or not pcm:
            raise AudioRejected("That file is not audio Scribe can read.")
        return pcm

    def _load(self):
        from faster_whisper import WhisperModel  # heavy import, kept out of module load

        return WhisperModel(
            self.model_name, device="cpu", compute_type="int8", download_root=str(self._models_dir)
        )

    def _run(self, pcm: bytes) -> Transcript:
        import numpy as np

        audio = np.frombuffer(pcm, np.int16).astype(np.float32) / 32768.0
        segments, _ = self._model.transcribe(audio, language="en", beam_size=1)
        parts = [Segment(round(s.start, 2), round(s.end, 2), s.text.strip()) for s in segments]
        return Transcript(" ".join(p.text for p in parts if p.text), parts, "en")

    async def transcribe(self, pcm: bytes) -> Transcript:
        try:
            async with self._lock:  # one model load, even for concurrent first calls
                if self._model is None:
                    self._model = await asyncio.to_thread(self._load)
            return await asyncio.to_thread(self._run, pcm)
        except Exception as error:  # model errors vary; never leak their text (may hold paths)
            raise EngineFailed("Transcription failed. Try again.") from error
```

```python
"""Upload and transcribe: validate, store, cache lookup, transcribe under a semaphore."""

from __future__ import annotations

import asyncio
import tempfile
from pathlib import Path

from src.config import Settings
from src.contract import SegmentOut, TranscribeResult
from src.engine import PCM_BYTES_PER_SECOND, SttEngine
from src.errors import AudioRejected, EngineFailed, FileNotFound, IdentityRequired
from src.store import AudioStore, StoredAudio


class ScribeService:
    def __init__(self, engine: SttEngine, store: AudioStore, settings: Settings) -> None:
        self._engine, self._store, self._settings = engine, store, settings
        self._slots = asyncio.Semaphore(settings.concurrency)
        self._locks: dict[tuple[str, str], asyncio.Lock] = {}

    async def upload(self, owner: str, filename: str, data: bytes) -> tuple[StoredAudio, float]:
        if not owner:
            raise IdentityRequired("Scribe needs to know who is asking.")
        duration = await self._probe(data)
        if duration > self._settings.max_seconds:
            raise AudioRejected(f"That audio is longer than {self._settings.max_seconds // 60} minutes.")
        file_id = AudioStore.file_id(owner, data)
        stored = await asyncio.to_thread(self._store.put, owner, file_id, data, filename, duration)
        return stored, self._store.expires_at(stored)

    async def _probe(self, data: bytes) -> float:
        """Decode a temporary copy: ffmpeg needs a seekable file for some formats (m4a)."""
        staging = self._settings.staging_dir
        await asyncio.to_thread(staging.mkdir, parents=True, exist_ok=True)

        def write() -> Path:
            handle = tempfile.NamedTemporaryFile(dir=staging, delete=False)
            with handle:
                handle.write(data)
            return Path(handle.name)

        path = await asyncio.to_thread(write)
        try:
            return len(await self._engine.decode(path)) / PCM_BYTES_PER_SECOND
        finally:
            await asyncio.to_thread(path.unlink, missing_ok=True)

    async def transcribe(self, owner: str, file_id: str, timestamps: bool) -> TranscribeResult:
        if not owner:
            raise IdentityRequired("Scribe needs to know who is asking.")
        key = (owner, file_id)
        lock = self._locks.setdefault(key, asyncio.Lock())
        try:
            async with lock:
                stored = await asyncio.to_thread(self._store.get, owner, file_id)
                if stored is None:
                    raise FileNotFound("That audio file expired or does not exist. Attach it again.")
                payload = await asyncio.to_thread(self._store.get_transcript, owner, file_id, self._engine.model_name)
                cached = payload is not None
                if payload is None:
                    payload = await self._run(owner, stored)
        finally:
            if not lock.locked() and not getattr(lock, "_waiters", None):
                self._locks.pop(key, None)
        segments = [SegmentOut(**s) for s in payload["segments"]] if timestamps else None
        return TranscribeResult(
            text=payload["text"], duration_s=stored.duration_s, language=payload["language"],
            segments=segments, cached=cached,
            message=f"Transcribed {stored.duration_s:.0f} s of audio ({len(payload['text'].split())} words).",
        )

    async def _run(self, owner: str, stored: StoredAudio) -> dict:
        pcm = await self._engine.decode(stored.path)
        try:
            async with self._slots:
                transcript = await asyncio.wait_for(self._engine.transcribe(pcm), self._settings.timeout_s)
        except TimeoutError as error:
            raise EngineFailed("Transcription took too long. Try a shorter file.") from error
        payload = {
            "text": transcript.text, "language": transcript.language,
            "segments": [{"start_s": s.start_s, "end_s": s.end_s, "text": s.text} for s in transcript.segments],
        }
        await asyncio.to_thread(self._store.put_transcript, owner, stored.file_id, self._engine.model_name, payload)
        return payload
```

```python
"""The Scribe server: one MCP tool, an audio upload route and a health check."""

from __future__ import annotations

from typing import Annotated

from mcp.server.fastmcp import Context, FastMCP
from mcp.server.fastmcp.exceptions import ToolError
from mcp.types import ToolAnnotations
from pydantic import Field
from starlette.applications import Starlette
from starlette.requests import Request
from starlette.responses import JSONResponse, Response

from src.config import Settings
from src.contract import TranscribeResult
from src.engine import SttEngine, WhisperEngine
from src.errors import AudioRejected, EngineFailed, IdentityRequired, QuotaExceeded, ScribeError
from src.identity import owner_from_context
from src.middleware import InternalTokenMiddleware
from src.service import ScribeService
from src.store import AudioStore

INSTRUCTIONS = (
    "Speech to text (English). The user attaches an audio file in chat; the question then contains a line "
    "'[Audio file_id: ...]'. Call tool_scribe_transcribe with that file_id to get the transcript."
)


def build_app(settings: Settings, engine: SttEngine | None = None) -> Starlette:
    engine = engine or WhisperEngine(settings.models_dir, settings.model, settings.ffmpeg)
    store = AudioStore(settings.audio_dir, settings.ttl_seconds, settings.quota_bytes)
    store.sweep()
    service = ScribeService(engine, store, settings)
    mcp = FastMCP(name="scribe", instructions=INSTRUCTIONS, host=settings.host, port=settings.port)

    @mcp.tool(
        meta={"keywords": ["transcribe", "speech to text", "stt", "audio", "dictation"], "display_label": "Transcribing audio"},
        annotations=ToolAnnotations(readOnlyHint=True, openWorldHint=False),
    )
    async def tool_scribe_transcribe(
        file_id: Annotated[str, Field(description="The file_id from the '[Audio file_id: ...]' line of the user's attachment.")],
        timestamps: Annotated[bool, Field(description="Also return per-segment start and end times.")] = False,
        ctx: Context = None,
    ) -> TranscribeResult:
        """Transcribe an attached English audio file to text. The file belongs to the asking user and expires
        after 24 hours. Returns the transcript, the duration and a summary."""
        try:
            return await service.transcribe(owner_from_context(ctx), file_id, timestamps)
        except ScribeError as error:
            raise ToolError(str(error)) from error

    @mcp.custom_route("/health", methods=["GET"])
    async def health(request: Request) -> Response:
        return JSONResponse({"ok": True})

    @mcp.custom_route("/api/files", methods=["POST"])
    async def upload(request: Request) -> Response:
        owner = request.headers.get("x-requester-username", "").strip()
        if not owner:
            return JSONResponse({"error": "identity_required"}, status_code=401)
        # shortcut: the whole body is read before the size check; fine behind ember_api's own cap, add streaming if exposed.
        data = await request.body()
        if len(data) > settings.max_bytes:
            return JSONResponse({"error": "too_large"}, status_code=413)
        # Only a display name: last path part, bounded; it never becomes a path.
        filename = request.headers.get("x-filename", "audio").replace("\\", "/").rsplit("/", 1)[-1][:255] or "audio"
        try:
            stored, expires_at = await service.upload(owner, filename, data)
        except QuotaExceeded as error:
            return JSONResponse({"error": error.code, "message": str(error)}, status_code=413)
        except AudioRejected as error:
            return JSONResponse({"error": error.code, "message": str(error)}, status_code=400)
        except EngineFailed as error:
            return JSONResponse({"error": error.code, "message": str(error)}, status_code=503)
        except IdentityRequired:
            return JSONResponse({"error": "identity_required"}, status_code=401)
        return JSONResponse({
            "file_id": stored.file_id, "duration_s": round(stored.duration_s, 1),
            "size_bytes": stored.size_bytes, "expires_at": expires_at,
        })

    app = mcp.streamable_http_app()
    app.add_middleware(InternalTokenMiddleware, token=settings.internal_token)
    return app
```

```python
"""Uploads an attached audio file to Scribe, which owns it for 24 hours.

Scribe checks the owner from the identity headers; the browser never learns
Scribe's origin or the internal token."""
from __future__ import annotations

import httpx
from fastapi import HTTPException

from src.models import Account
from src.services.mcp_session import identity_headers


class ScribeGateway:
    def __init__(self, client: httpx.AsyncClient, base_url: str, token: str | None):
        self.client, self.base, self.token = client, base_url.rstrip("/"), token

    async def upload(self, account: Account, filename: str, content: bytes) -> dict:
        headers = identity_headers(account.username, account.email, self.token, account.uid)
        headers["X-Filename"] = filename
        headers["Content-Type"] = "application/octet-stream"
        try:
            response = await self.client.post(f"{self.base}/api/files", headers=headers, content=content, timeout=120)
        except httpx.HTTPError as error:
            raise HTTPException(502, "Scribe is unreachable. Try again.") from error
        if response.status_code == 200:
            return response.json()
        if response.status_code in (400, 413):
            try:
                message = response.json().get("message")
            except ValueError:
                message = None
            raise HTTPException(response.status_code if response.status_code == 413 else 400, message or "Scribe rejected the audio.")
        raise HTTPException(502, "Scribe is unavailable; check its internal connection.")
```

```python
# 20 MB, not Scribe's 25 MB: base64 of 25 MB would not fit ember_api's request cap for this path.
_MAX_AUDIO_BYTES = 20 * 1024 * 1024
_MAX_AUDIO_BASE64 = (_MAX_AUDIO_BYTES + 2) // 3 * 4


class AudioIn(BaseModel):
    filename: str = Field(min_length=1, max_length=255)
    data: str = Field(min_length=1, max_length=_MAX_AUDIO_BASE64)


class AttachmentAudioOut(BaseModel):
    filename: str
    file_id: str
    duration_s: float
    expires_at: float


@router.post("/audio")
async def attachment_audio(
    body: AudioIn,
    request: Request,
    account: Account = Depends(require_chat),
    logs: LogWriter = Depends(get_log_writer),
) -> AttachmentAudioOut:
    """Keep the original for Scribe's transcribe tool; the agent gets the file_id."""
    filename = body.filename.replace("\\", "/").rsplit("/", 1)[-1]
    if not filename.lower().endswith(_AUDIO_SUFFIXES):
        raise HTTPException(400, "Upload an mp3, wav, m4a, ogg, flac or webm file")
    try:

    uploaded = await gateway.upload(account, filename, content)
    await logs.action(account, "attachments.audio", f"Attached '{filename}' ({len(content):,} bytes) for transcription")
    return AttachmentAudioOut(
```
