# Bard (text to speech) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A standalone text-to-speech MCP service (Bard, port 8090) that Armory proxies as an extension, with an ember_api audio route and an in-chat player in ember_web.

**Architecture:** Bard is a FastMCP server with two tools plus a token-protected HTTP file route. Piper synthesizes on a worker thread, ffmpeg encodes MP3, and an owner-scoped store caches files with a time-to-live and a quota. Armory forwards the asking user's name; ember_api streams stored audio to the browser because ember_web's CSP only allows same-origin media.

**Tech Stack:** Python 3.11+ (3.14 on this machine), `mcp==1.28.0` (FastMCP), `piper-tts==1.8.0`, uvicorn, Starlette, ffmpeg; ember_api (FastAPI, httpx); ember_web (Vue 3, TypeScript, vitest).

**Spec:** `docs/superpowers/specs/2026-10-25-bard-design.md` (relative to `Python/Citadel/`)

**Commits:** the commit steps below run only when the user has asked for commits. Bard commits go in its own repo (`Python/Bard`); Armory, ember_api and ember_web commits go in the Citadel repo (`Python/Citadel`). Never run git at `D:\User\Documents\Programming`.

## Global Constraints

- Bard listens on port `8090`, MCP endpoint `http://127.0.0.1:8090/mcp`, extension id `bard`; tools are exactly `tool_bard_speak` and `tool_bard_listVoices`.
- Default voice `en_US-lessac-medium`; output format MP3 only.
- Every blocking call (Piper, ffmpeg wait, file IO) stays off the event loop (`asyncio.to_thread` or asyncio subprocess).
- Result models end with a `message: str` field; tools never call an AI; error messages never contain filesystem paths.
- File id is `sha256(voice + "\0" + prepared text)[:32]` hex, checked against `^[0-9a-f]{32}$` before any path is built.
- Owner key is the username from `_meta.requester.username`; missing identity fails with `identity_required`.
- `INTERNAL_API_TOKEN` guards `/mcp` and `/api/*` (constant-time compare); a non-loopback `BARD_HOST` with an empty token refuses to start.
- ember_web: no reduced-motion setting, media query or test. Audio is always played through `/api/bard/audio/{file_id}` (same origin).
- Never read or print `.env` files. Real `configs/config_extensions.json` is operator-owned: change only the `.example`.
- Commands run from the folder named in each step. Bard: `.venv_bard/Scripts/python.exe`; Armory: `.venv_mcp/Scripts/python.exe`; ember_api: `.venv_ember_api/Scripts/python.exe`.

```python
"""Turn chat Markdown into plain speakable text."""

from __future__ import annotations

import re

from src.errors import TextRejected

_FENCE = re.compile(r"```.*?```", re.S)
_LINK = re.compile(r"!?\[([^\]]*)\]\([^)]*\)")
_INLINE_CODE = re.compile(r"`([^`]*)`")
_BLOCK = re.compile(r"(?m)^\s{0,3}(?:#{1,6}|>|[-*+]|\d+\.)\s+")
_EMPHASIS = re.compile(r"\*{1,3}|~~|(?<!\w)_+|_+(?!\w)")
_SPACE = re.compile(r"\s+")


def prepare(text: str, max_chars: int) -> str:
    cleaned = _FENCE.sub(" ", text)
    cleaned = _LINK.sub(r"\1", cleaned)
    cleaned = _INLINE_CODE.sub(r"\1", cleaned)
    cleaned = _BLOCK.sub("", cleaned)
    cleaned = _EMPHASIS.sub("", cleaned)
    cleaned = _SPACE.sub(" ", cleaned).strip()
    if not cleaned:
        raise TextRejected("There is no speakable text.")
    if len(cleaned) > max_chars:
        raise TextRejected(f"The text is {len(cleaned)} characters; the limit is {max_chars}.")
    return cleaned
```

`src/store.py`:

```python
"""Per-owner MP3 files with a JSON sidecar, a time-to-live and a quota.

Blocking by design (plain file IO); the service calls it through asyncio.to_thread.
"""

from __future__ import annotations

import hashlib
import json
import os
import re
import time
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from src.errors import QuotaExceeded

_ID = re.compile(r"^[0-9a-f]{32}$")


@dataclass(frozen=True)
class StoredAudio:
    file_id: str
    path: Path
    voice: str
    duration_s: float
    size_bytes: int
    created_at: float


class AudioStore:
    def __init__(
        self,
        root: Path,
        ttl_seconds: int,
        quota_bytes: int,
        clock: Callable[[], float] = time.time,
    ) -> None:
        self._root = Path(root)
        self._ttl = ttl_seconds
        self._quota = quota_bytes
        self._clock = clock

    @staticmethod
    def file_id(voice: str, text: str) -> str:
        return hashlib.sha256(f"{voice}\0{text}".encode("utf-8")).hexdigest()[:32]

    def _owner_dir(self, owner: str) -> Path:
        return self._root / hashlib.sha256(owner.encode("utf-8")).hexdigest()[:16]

    def _expired(self, stored: StoredAudio) -> bool:
        return stored.created_at + self._ttl <= self._clock()

    @staticmethod
    def _read(owner_dir: Path, file_id: str) -> StoredAudio | None:
        mp3 = owner_dir / f"{file_id}.mp3"
        try:
            info = json.loads((owner_dir / f"{file_id}.json").read_text(encoding="utf-8"))
            return StoredAudio(
                file_id, mp3, info["voice"], float(info["duration_s"]), mp3.stat().st_size, float(info["created_at"])
            )
        except (OSError, ValueError, KeyError):
            return None

    @staticmethod
    def _remove(owner_dir: Path, file_id: str) -> None:
        for suffix in (".mp3", ".json"):
            (owner_dir / f"{file_id}{suffix}").unlink(missing_ok=True)

    @staticmethod
    def _write(path: Path, data: bytes) -> None:
        tmp = path.with_name(path.name + ".tmp")
        tmp.write_bytes(data)
        os.replace(tmp, path)

    def get(self, owner: str, file_id: str) -> StoredAudio | None:
        if not _ID.match(file_id):
            return None
        owner_dir = self._owner_dir(owner)
        stored = self._read(owner_dir, file_id)
        if stored is None:
            return None
        if self._expired(stored):
            self._remove(owner_dir, file_id)
            return None
        return stored

    def _sweep_dir(self, owner_dir: Path) -> int:
        removed = 0
        for sidecar in owner_dir.glob("*.json"):
            stored = self._read(owner_dir, sidecar.stem)
            if stored is None or self._expired(stored):
                self._remove(owner_dir, sidecar.stem)
                removed += 1
        return removed

    def sweep(self) -> int:
        if not self._root.exists():
            return 0
        return sum(self._sweep_dir(d) for d in self._root.iterdir() if d.is_dir())

    def put(self, owner: str, file_id: str, mp3: bytes, voice: str, duration_s: float) -> StoredAudio:
        owner_dir = self._owner_dir(owner)
        owner_dir.mkdir(parents=True, exist_ok=True)
        self._sweep_dir(owner_dir)
        used = sum(p.stat().st_size for p in owner_dir.glob("*.mp3"))
        if used + len(mp3) > self._quota:
            raise QuotaExceeded("Your audio storage is full. Wait for older audio to expire and try again.")
        created = self._clock()
        meta = {"voice": voice, "duration_s": duration_s, "created_at": created}
        self._write(owner_dir / f"{file_id}.json", json.dumps(meta).encode("utf-8"))
        self._write(owner_dir / f"{file_id}.mp3", mp3)
        return StoredAudio(file_id, owner_dir / f"{file_id}.mp3", voice, duration_s, len(mp3), created)
```

`src/engine.py`:

```python
"""Text to MP3. Piper is CPU-bound and blocking, so it runs on a worker thread;
ffmpeg runs as an asyncio subprocess."""

from __future__ import annotations

import asyncio
import io
import threading
import wave
from dataclasses import dataclass
from pathlib import Path
from typing import Protocol

from src.errors import EngineFailed, VoiceUnavailable


@dataclass(frozen=True)
class Synthesis:
    mp3: bytes
    duration_s: float


class TtsEngine(Protocol):
    def voices(self) -> list[str]: ...

    async def synthesize(self, text: str, voice: str) -> Synthesis: ...


class PiperEngine:
    def __init__(self, voices_dir: Path, ffmpeg: str = "ffmpeg") -> None:
        self._dir = Path(voices_dir)
        self._ffmpeg = ffmpeg
        self._loaded: dict[str, object] = {}
        self._load_lock = threading.Lock()

    def voices(self) -> list[str]:
        return sorted(p.name.removesuffix(".onnx") for p in self._dir.glob("*.onnx")) if self._dir.is_dir() else []

    def _voice(self, name: str):
        with self._load_lock:
            if name not in self._loaded:
                path = self._dir / f"{name}.onnx"
                if not path.is_file():
                    raise VoiceUnavailable(f"Voice '{name}' is not installed.")
                from piper import PiperVoice  # heavy import, only when a voice is first used

                self._loaded[name] = PiperVoice.load(path)
            return self._loaded[name]

    def _wav(self, text: str, name: str) -> tuple[bytes, float]:
        voice = self._voice(name)
        buf = io.BytesIO()
        with wave.open(buf, "wb") as wav:
            voice.synthesize_wav(text, wav)
            seconds = wav.getnframes() / wav.getframerate()
        return buf.getvalue(), seconds

    async def _mp3(self, wav: bytes) -> bytes:
        try:
            proc = await asyncio.create_subprocess_exec(
                self._ffmpeg, "-loglevel", "error", "-f", "wav", "-i", "pipe:0",
                "-codec:a", "libmp3lame", "-q:a", "4", "-f", "mp3", "pipe:1",
                stdin=asyncio.subprocess.PIPE, stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
            )
        except FileNotFoundError as error:
            raise EngineFailed("ffmpeg was not found on the Bard host.") from error
        out, err = await proc.communicate(wav)
        if proc.returncode != 0:
            raise EngineFailed(f"ffmpeg could not encode the audio: {err.decode('utf-8', 'replace').strip()[:200]}")
        return out

    async def synthesize(self, text: str, voice: str) -> Synthesis:
        wav, seconds = await asyncio.to_thread(self._wav, text, voice)
        return Synthesis(await self._mp3(wav), seconds)
```

```python
"""Speak: prepare the text, reuse a stored file or synthesize one, store it."""

from __future__ import annotations

import asyncio

from src.config import Settings
from src.contract import SpeakResult, VoiceList
from src.engine import TtsEngine
from src.errors import IdentityRequired, VoiceUnavailable
from src.store import AudioStore
from src.text_prep import prepare


class BardService:
    def __init__(self, engine: TtsEngine, store: AudioStore, settings: Settings) -> None:
        self._engine = engine
        self._store = store
        self._settings = settings
        self._slots = asyncio.Semaphore(settings.max_concurrent)
        # shortcut: a lock may be dropped while a waiter still holds it, so two identical
        # requests can then both synthesize; harmless (same file id), upgrade if it ever hurts.
        self._locks: dict[tuple[str, str], asyncio.Lock] = {}

    async def list_voices(self) -> VoiceList:
        voices = await asyncio.to_thread(self._engine.voices)
        return VoiceList(
            voices=voices,
            default=self._settings.voice,
            message=f"{len(voices)} voice(s) installed; the default is {self._settings.voice}.",
        )

    async def speak(self, owner: str, text: str, voice: str | None = None) -> SpeakResult:
        if not owner:
            raise IdentityRequired("No signed-in user was sent with this request.")
        voice = voice or self._settings.voice
        if voice not in await asyncio.to_thread(self._engine.voices):
            raise VoiceUnavailable(f"Voice '{voice}' is not installed. Call tool_bard_listVoices for the list.")
        clean = prepare(text, self._settings.max_chars)
        file_id = AudioStore.file_id(voice, clean)
        key = (owner, file_id)
        lock = self._locks.setdefault(key, asyncio.Lock())
        try:
            async with lock:
                stored = await asyncio.to_thread(self._store.get, owner, file_id)
                cached = stored is not None
                if stored is None:
                    async with self._slots:
                        synthesis = await self._engine.synthesize(clean, voice)
                    stored = await asyncio.to_thread(
                        self._store.put, owner, file_id, synthesis.mp3, voice, synthesis.duration_s
                    )
        finally:
            if self._locks.get(key) is lock and not lock.locked():
                del self._locks[key]
        note = ", from cache" if cached else ""
        return SpeakResult(
            file_id=file_id,
            voice=voice,
            duration_s=stored.duration_s,
            size_bytes=stored.size_bytes,
            cached=cached,
            message=f"Speech ready ({stored.duration_s:.1f} s, voice {voice}{note}). The player is shown in the chat.",
        )
```

```python
"""Require the shared internal token on /mcp and /api. Only enforced when a token is configured."""

from __future__ import annotations

import hmac
import json

TOKEN_HEADER = b"x-internal-token"


def _protected(path: str) -> bool:
    return path == "/mcp" or path.startswith("/mcp/") or path.startswith("/api/")


class InternalTokenMiddleware:
    def __init__(self, app, token: str) -> None:
        self.app = app
        self._token = token.encode("utf-8")

    async def __call__(self, scope, receive, send):
        if self._token and scope["type"] == "http" and _protected(scope.get("path", "")):
            provided = dict(scope.get("headers") or []).get(TOKEN_HEADER, b"")
            if not hmac.compare_digest(self._token, provided):
                body = json.dumps({"error": "Invalid or missing internal API token"}).encode("utf-8")
                await send({
                    "type": "http.response.start",
                    "status": 401,
                    "headers": [(b"content-type", b"application/json"), (b"content-length", str(len(body)).encode())],
                })
                await send({"type": "http.response.body", "body": body})
                return
        await self.app(scope, receive, send)
```

```python
"""The Bard server: two MCP tools, an audio file route and a health check."""

from __future__ import annotations

import asyncio
from typing import Annotated

from mcp.server.fastmcp import Context, FastMCP
from mcp.server.fastmcp.exceptions import ToolError
from mcp.types import ToolAnnotations
from pydantic import Field
from starlette.applications import Starlette
from starlette.requests import Request
from starlette.responses import FileResponse, JSONResponse, Response

from src.config import Settings
from src.contract import SpeakResult, VoiceList
from src.engine import PiperEngine, TtsEngine
from src.errors import BardError
from src.identity import owner_from_context
from src.middleware import InternalTokenMiddleware
from src.service import BardService
from src.store import AudioStore

INSTRUCTIONS = (
    "Text to speech. Call tool_bard_speak with the text to say; the chat shows a player and a download link. "
    "Pass only text a user should hear. Call tool_bard_listVoices to see the installed voices."
)


def build_app(settings: Settings, engine: TtsEngine | None = None) -> Starlette:
    engine = engine or PiperEngine(settings.voices_dir, settings.ffmpeg)
    store = AudioStore(settings.audio_dir, settings.ttl_seconds, settings.quota_bytes)
    store.sweep()
    service = BardService(engine, store, settings)
    mcp = FastMCP(name="bard", instructions=INSTRUCTIONS, host=settings.host, port=settings.port)

    @mcp.tool(meta={"keywords": ["speak", "say", "read aloud", "tts", "voice", "audio"], "display_label": "Generating speech"})
    async def tool_bard_speak(
        text: Annotated[str, Field(description="The text to speak. Markdown is stripped; code blocks are skipped.")],
        voice: Annotated[str | None, Field(description="An installed voice name. Omit it for the default voice.")] = None,
        ctx: Context = None,
    ) -> SpeakResult:
        """Turn text into spoken English (MP3). Creates an audio file owned by the asking user; the chat shows
        a player and a download link, so do not paste a link yourself. Returns the file id, duration and a summary."""
        try:
            return await service.speak(owner_from_context(ctx), text, voice)
        except BardError as error:
            raise ToolError(str(error)) from error

    @mcp.tool(
        meta={"keywords": ["voices", "tts"], "display_label": "Listing voices"},
        annotations=ToolAnnotations(readOnlyHint=True, openWorldHint=False),
    )
    async def tool_bard_listVoices() -> VoiceList:
        """List the installed voices and the default one. Read-only."""
        return await service.list_voices()

    @mcp.custom_route("/health", methods=["GET"])
    async def health(request: Request) -> Response:
        return JSONResponse({"ok": True})

    @mcp.custom_route("/api/files/{file_id}", methods=["GET"])
    async def audio(request: Request) -> Response:
        owner = request.headers.get("x-requester-username", "").strip()
        if not owner:
            return JSONResponse({"error": "identity_required"}, status_code=401)
        file_id = request.path_params["file_id"]
        stored = await asyncio.to_thread(store.get, owner, file_id)
        if stored is None:
            return JSONResponse({"error": "file_not_found"}, status_code=404)
        return FileResponse(
            stored.path,
            media_type="audio/mpeg",
            headers={
                "Cache-Control": "private, no-store",
                "X-Content-Type-Options": "nosniff",
                "Content-Disposition": f'inline; filename="bard-{file_id[:8]}.mp3"',
            },
        )

    app = mcp.streamable_http_app()
    app.add_middleware(InternalTokenMiddleware, token=settings.internal_token)
    return app
```

```python
"""Streams a stored Bard audio file to the signed-in browser.

Bard owns the files and checks the owner from the identity headers; the browser
never learns Bard's origin or the internal token."""
from __future__ import annotations

from urllib.parse import quote

import httpx
from fastapi import HTTPException
from fastapi.responses import StreamingResponse

from src.models import Account
from src.services.mcp_session import identity_headers

_FORWARDED = ("content-type", "content-length", "content-range", "accept-ranges")


class BardGateway:
    def __init__(self, client: httpx.AsyncClient, base_url: str, token: str | None):
        self.client, self.base, self.token = client, base_url.rstrip("/"), token

    async def stream(self, file_id: str, account: Account, range_header: str | None = None) -> StreamingResponse:
        headers = identity_headers(account.username, account.email, self.token, account.uid)
        headers["Accept-Encoding"] = "identity"
        if range_header:
            headers["Range"] = range_header
        request = self.client.build_request(
            "GET", f"{self.base}/api/files/{quote(file_id, safe='')}", headers=headers, timeout=60
        )
        try:
            response = await self.client.send(request, stream=True, follow_redirects=False)
        except httpx.HTTPError as error:
            raise HTTPException(502, "Bard is unreachable. Try again.") from error
        if response.status_code not in (200, 206):
            await response.aread()
            await response.aclose()
            if response.status_code in (404, 416):
                raise HTTPException(response.status_code, "The audio expired or is unavailable.")
            raise HTTPException(502, "Bard is unavailable; check its internal connection.")

        async def chunks():
            try:
                async for chunk in response.aiter_bytes():
                    yield chunk
            finally:
                await response.aclose()

        forwarded = {k: v for k, v in response.headers.items() if k.lower() in _FORWARDED}
        forwarded.update({
            "Cache-Control": "no-store",
            "X-Content-Type-Options": "nosniff",
            "Referrer-Policy": "no-referrer",
            "Content-Disposition": 'inline; filename="bard.mp3"',
        })
        if response.headers.get("content-encoding"):
            forwarded.pop("content-length", None)
        return StreamingResponse(chunks(), status_code=response.status_code, headers=forwarded)
```

`src/utils/tools/bardAudio.ts`:

```ts
import type { ToolStep } from "../../api/types";

// Matched by suffix: an extension's tools arrive namespaced (`bard__tool_bard_speak`).
const SPEAK_TOOL = "tool_bard_speak";

/** File ids of the audio Bard produced in these steps, in order and without repeats. */
export function bardFileIds(steps: readonly ToolStep[] | undefined): string[] {
  const ids: string[] = [];
  for (const step of steps ?? []) {
    if (step.ok !== true || !step.tool.endsWith(SPEAK_TOOL)) continue;
    try {
      const id = (JSON.parse(step.result) as { file_id?: unknown }).file_id;
      if (typeof id === "string" && !ids.includes(id)) ids.push(id);
    } catch {
      // Not JSON: nothing to play.
    }
  }
  return ids;
}
```
